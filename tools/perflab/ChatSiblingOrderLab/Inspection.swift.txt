
extension ChatLayerNode {
    func sortChildrenForLab() { sortChildren() }
    var orderedChildrenForLab: [ChatLayerNode] { orderedChildren }
    var orderingSharesStorageForLab: Bool {
        children.withUnsafeBufferPointer { a in
            orderedChildren.withUnsafeBufferPointer { b in
                !a.isEmpty && a.baseAddress == b.baseAddress
            }
        }
    }
}
