    public var signature: [String] {
        var result = ["size:\(layoutSize)", "perfect:\(isPerfectSized)", "emoji:\(isBigEmoji)"]
        for line in lines {
            result.append("line:\(line.frame):\(line.range):\(line.penFlush):\(line.isRTL):\(line.isBlocked)")
            for index in line.range.location ... line.range.location + line.range.length {
                result.append("offset:\(CTLineGetOffsetForStringIndex(line.line, index, nil))")
            }
            for x in stride(from: CGFloat(-5), through: line.frame.width + 5, by: 9) {
                result.append("hit:\(CTLineGetStringIndexForPosition(line.line, CGPoint(x: x, y: 0)))")
            }
            for value in line.strikethrough { result.append("strike:\(value.frame):\(value.color)") }
            for value in line.underlines { result.append("under:\(value.frame):\(value.color)") }
        }
        for rect in toolTipRects { result.append("tooltip:\(rect)") }
        for value in strokeRects { result.append("stroke:\(value.0):\(value.1)") }
        for value in hexColorsRect { result.append("hex:\(value.0):\(value.1):\(value.2)") }
        for value in embeddedItems { result.append("embed:\(value.range):\(value.rect):\(value.value)") }
        for value in blockQuotes { result.append("quote:\(value.frame):\(value.range):\(value.isCode)") }
        return result
    }
    public func pixels() -> Data {
        let width = max(1, Int(ceil(layoutSize.width))+10)
        let height = max(1, Int(ceil(layoutSize.height))+20)
        let context = CGContext(data: nil, width: width, height: height, bitsPerComponent: 8, bytesPerRow: width*4, space: CGColorSpaceCreateDeviceRGB(), bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue)!
        context.setFillColor(NSColor.white.cgColor)
        context.fill(CGRect(x: 0, y: 0, width: width, height: height))
        for line in lines {
            context.textPosition = CGPoint(x: line.frame.minX + CGFloat(CTLineGetPenOffsetForFlush(line.line, line.penFlush, Double(layoutSize.width))), y: CGFloat(height)-line.frame.minY-10)
            CTLineDraw(line.line, context)
        }
        return Data(bytes: context.data!, count: width*height*4)
    }
