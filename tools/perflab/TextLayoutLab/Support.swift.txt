public extension NSAttributedString {
    var range: NSRange { NSRange(location: 0, length: length) }
    func containsAttribute(attributeName: NSAttributedString.Key) -> Any? {
        var result: Any?
        enumerateAttribute(attributeName, in: range) { value, _, stop in
            if let value { result = value; stop.pointee = true }
        }
        return result
    }
}
public extension NSRange {
    var min: Int { location }
    var max: Int { location + length }
    var isEmpty: Bool { length == 0 }
    func intersection(_ other: NSRange) -> NSRange? {
        let value = NSIntersectionRange(self, other)
        return value.length > 0 ? value : nil
    }
}

public func input(_ text: String, style: Int = 0) -> NSAttributedString {
    let result = NSMutableAttributedString(string: text, attributes: [.font: NSFont.systemFont(ofSize: 13), .foregroundColor: NSColor.black])
    let length = result.length
    guard length > 0 else { return result }
    switch style {
    case 1:
        for i in stride(from: 0, to: length, by: 23) {
            result.addAttribute(.link, value: "https://example.com/\(i)", range: NSRange(location: i, length: min(10, length-i)))
        }
    case 2:
        result.addAttribute(.strikethroughStyle, value: 1, range: NSRange(location: 0, length: length/2))
        result.addAttribute(.underlineStyle, value: 1, range: NSRange(location: length/2, length: length-length/2))
    case 3:
        result.addAttribute(TextInputAttributes.quote, value: TextViewBlockQuoteData(id: 1, colors: .init(main: .blue)), range: NSRange(location: 0, length: length/2))
    case 4:
        for i in stride(from: 0, to: length, by: 29) {
            let range = NSRange(location: i, length: min(6, length-i))
            result.addAttribute(.hexColorMark, value: NSColor.red, range: range)
            result.addAttribute(.hexColorMarkDimensions, value: NSSize(width: 18, height: 18), range: range)
        }
    case 5:
        for i in stride(from: 0, to: length, by: 19) {
            result.addAttribute(TextInputAttributes.embedded, value: i, range: NSRange(location: i, length: 1))
        }
    case 6:
        for i in stride(from: 0, to: length, by: 17) {
            result.addAttributes([.font: NSFont.systemFont(ofSize: CGFloat(9+i%20), weight: .bold), .foregroundColor: NSColor.red], range: NSRange(location: i, length: min(8,length-i)))
        }
    default: break
    }
    return result
}
