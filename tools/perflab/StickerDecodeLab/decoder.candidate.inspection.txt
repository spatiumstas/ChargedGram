
func decodeFallbackForLab(_ data: Data) -> String {
    if data.isEmpty { return "Mz" }
    return data.withUnsafeBytes { (input: UnsafeRawBufferPointer) in
        stickerThumbnailPathFallback(input, length: stickerThumbnailPathLength(input))
    }
}
func decodeLengthForLab(_ data: Data) -> Int {
    data.withUnsafeBytes { (input: UnsafeRawBufferPointer) in stickerThumbnailPathLength(input) }
}
