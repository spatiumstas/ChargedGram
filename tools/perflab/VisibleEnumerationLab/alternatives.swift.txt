
extension TableView {
    public func enumerationLabCopied(reversed: Bool = false, inset: CGFloat = 0, with callback: (TableRowItem) -> Bool) {
        let visible = visibleRows(inset)
        guard visible.length > 0, visible.location >= 0, visible.location < list.count else { return }
        let end = visible.location + min(visible.length, list.count - visible.location)
        let items = Array(list[visible.location..<end])
        if reversed {
            for item in items.reversed() {
                if !callback(item) { break }
            }
        } else {
            for item in items {
                if !callback(item) { break }
            }
        }
    }

    public func enumerationLabTemporary(reversed: Bool = false, inset: CGFloat = 0, with callback: (TableRowItem) -> Bool) {
        let visible = visibleRows(inset)
        guard visible.length > 0, visible.location >= 0, visible.location < list.count else { return }
        let count = min(visible.length, list.count - visible.location)
        withUnsafeTemporaryAllocation(of: TableRowItem.self, capacity: count) { buffer in
            _ = buffer.initialize(from: list[visible.location..<(visible.location + count)])
            defer { buffer.deinitialize() }
            if reversed {
                for item in buffer.reversed() {
                    if !callback(item) { break }
                }
            } else {
                for item in buffer {
                    if !callback(item) { break }
                }
            }
        }
    }
}

extension TableView {
    public func enumerationLabBulk(reversed: Bool = false, inset: CGFloat = 0, with callback: (TableRowItem) -> Bool) {
        let visible = visibleRows(inset)
        guard visible.length > 0, visible.location >= 0, visible.location < list.count else { return }
        let count = min(visible.length, list.count - visible.location)
        if count == 1 {
            _ = callback(list[visible.location])
            return
        }
        withUnsafeTemporaryAllocation(of: TableRowItem.self, capacity: count) { buffer in
            list.withUnsafeBufferPointer { source in
                buffer.baseAddress!.initialize(from: source.baseAddress!.advanced(by: visible.location), count: count)
            }
            defer { buffer.deinitialize() }
            if reversed {
                for item in buffer.reversed() {
                    if !callback(item) { break }
                }
            } else {
                for item in buffer {
                    if !callback(item) { break }
                }
            }
        }
    }
}
