
extension TextViewLayout {
    public static func quoteFixture(_ string: String, pattern: [Int], decorated: Bool = false) -> NSMutableAttributedString {
        let result = NSMutableAttributedString(string: string, attributes: [.font: NSFont.systemFont(ofSize: 14), .foregroundColor: NSColor.black])
        let header = TextNode.layoutText(NSAttributedString(string: "Swift", attributes: [.font: NSFont.systemFont(ofSize: 12), .foregroundColor: NSColor.blue]), nil, 1, .end, NSSize(width: 200, height: 20), nil, false, .left)
        let quotes = [
            TextViewBlockQuoteData(id: 1, colors: .init(main: .blue), collapsable: true),
            TextViewBlockQuoteData(id: 2, colors: .init(main: .red), isCode: true, space: 6, header: header)
        ]
        for index in 0..<result.length {
            let code = pattern.isEmpty ? 0 : pattern[index % pattern.count]
            if code == 1 || code == 2 {
                result.addAttribute(TextInputAttributes.quote, value: quotes[code - 1], range: NSRange(location: index, length: 1))
            } else if code == 3 {
                result.addAttribute(TextInputAttributes.quote, value: NSNumber(value: 42), range: NSRange(location: index, length: 1))
            }
            if decorated {
                result.addAttribute(.foregroundColor, value: index % 2 == 0 ? NSColor.blue : NSColor.red, range: NSRange(location: index, length: 1))
                if index % 7 == 0 { result.addAttribute(.underlineStyle, value: 1, range: NSRange(location: index, length: 1)) }
                if index % 11 == 0 { result.addAttribute(.strikethroughStyle, value: 1, range: NSRange(location: index, length: 1)) }
            }
        }
        return result
    }

    public func quoteLookupIDs(_ range: NSRange) -> [Int] {
        var last: Any?
        attributedString.enumerateAttribute(TextInputAttributes.quote, in: range) { value, _, _ in
            if let value { last = value }
        }
        return [(last as? TextViewBlockQuoteData)?.id ?? -1, lastBlockQuote(in: range)?.id ?? -1]
    }

    public func quoteMeasurementSnapshot() -> [Double] {
        var result = [Double(layoutSize.width), Double(layoutSize.height), Double(perfectSize.width), Double(perfectSize.height), isPerfectSized ? 1.0 : 0, Double(lines.count), Double(blockQuotes.count)]
        func rect(_ value: CGRect) { result += [Double(value.minX), Double(value.minY), Double(value.width), Double(value.height)] }
        for entry in lines {
            rect(entry.frame)
            result += [Double(entry.range.location), Double(entry.range.length), Double(entry.lineRange.location), Double(entry.lineRange.length), Double(entry.penFlush), entry.isRTL ? 1 : 0, entry.isBlocked ? 1 : 0]
            let runs = CTLineGetGlyphRuns(entry.line) as! [CTRun]
            result.append(Double(runs.count))
            for run in runs {
                let count = CTRunGetGlyphCount(run)
                var glyphs = [CGGlyph](repeating: 0, count: count)
                var points = [CGPoint](repeating: .zero, count: count)
                var advances = [CGSize](repeating: .zero, count: count)
                var indexes = [CFIndex](repeating: 0, count: count)
                CTRunGetGlyphs(run, CFRange(location: 0, length: 0), &glyphs)
                CTRunGetPositions(run, CFRange(location: 0, length: 0), &points)
                CTRunGetAdvances(run, CFRange(location: 0, length: 0), &advances)
                CTRunGetStringIndices(run, CFRange(location: 0, length: 0), &indexes)
                result.append(Double(count))
                for index in 0..<count {
                    result += [Double(glyphs[index]), Double(points[index].x), Double(points[index].y), Double(advances[index].width), Double(advances[index].height), Double(indexes[index])]
                }
            }
            result += [Double(entry.strikethrough.count), Double(entry.underlines.count), Double(entry.embeddedItems.count)]
            for value in entry.strikethrough { rect(value.frame) }
            for value in entry.underlines { rect(value.frame) }
            for value in entry.embeddedItems { rect(value.frame) }
        }
        for block in blockQuotes {
            rect(block.frame)
            result += [Double(block.range.location), Double(block.range.length), block.isCode ? 1 : 0, Double(block.headerInset), block.colors.main == NSColor.blue ? 1 : 2]
            if let header = block.header { result += [Double(header.0.size.width), Double(header.0.size.height)] }
        }
        return result
    }

    public func quotePixels(scale: Int) -> Data {
        let width = max(1, Int(ceil(layoutSize.width)) + 16)
        let height = max(1, Int(ceil(layoutSize.height)) + 16)
        let context = CGContext(data: nil, width: width * scale, height: height * scale, bitsPerComponent: 8, bytesPerRow: width * scale * 4, space: CGColorSpaceCreateDeviceRGB(), bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue)!
        context.scaleBy(x: CGFloat(scale), y: CGFloat(scale))
        let layer = TextViewLayer()
        layer.contentsScale = CGFloat(scale)
        layer.update(self)
        layer.draw(in: context)
        return Data(bytes: context.data!, count: context.bytesPerRow * context.height)
    }
}
