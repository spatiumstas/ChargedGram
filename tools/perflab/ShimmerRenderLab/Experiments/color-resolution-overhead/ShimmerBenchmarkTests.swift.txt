import Cocoa
import XCTest
import Darwin
import ObjectiveC
import enum TGUIKit.ShimmerRasterProbe
import enum TGUIKitBaseline.ShimmerRasterProbe
import struct TGUIKit.System
import struct TGUIKitBaseline.System

final class ShimmerBenchmarkTests: XCTestCase {
    func testCosts() throws {
        guard ProcessInfo.processInfo.environment["SHIMMER_RENDER_BENCHMARK"] == "1" else { throw XCTSkip("Opt-in shimmer render timing") }
        MeasureBootstrap.install()
        TGUIKit.System.updateScaleFactor(1); TGUIKitBaseline.System.updateScaleFactor(1)
        let reverse = ProcessInfo.processInfo.environment["SHIMMER_RENDER_REVERSE"] == "1"
        let factory = ShimmerTests()
        TGUIKitBaseline.ShimmerRasterProbe.configure(before: nil)
        TGUIKit.ShimmerRasterProbe.configure(before: nil)
        defer {
            TGUIKitBaseline.ShimmerRasterProbe.configure(before: nil)
            TGUIKit.ShimmerRasterProbe.configure(before: nil)
        }
        var detailed = Data([0, 128])
        for i in 0..<60 { detailed.append(contentsOf: [203, UInt8((i * 17) % 64), UInt8(128 + (i * 31) % 64)]) }
        var results: [String: [[Double]]] = [:], work: [String: [Int]] = [:], sizes: [String: [Int]] = [:]
        for kind in ShimmerKind.allCases {
            let objects = [shimmer(kind, baseline: true), shimmer(kind, baseline: false)]
            sizes[String(describing: kind) + "_object_allocation"] = objects.map { malloc_size(Unmanaged.passUnretained($0.object).toOpaque()) }
            sizes[String(describing: kind) + "_runtime_instance"] = objects.map { class_getInstanceSize(object_getClass($0.object)!) }
            for hasData in [false, true] {
                for transparent in [false, true] {
                    if kind == .silhouette && !transparent { continue }
                    var input = ShimmerInput(); input.data = hasData ? detailed : nil; input.background = transparent ? nil : .white
                    let previous = objects.map { $0.image() }
                    for object in objects { object.update(input) }
                    factory.until { objects.enumerated().allSatisfy { $0.element.image() != nil && $0.element.image() !== previous[$0.offset] } }
                    factory.pump()
                    let count = kind == .sticker ? 1000 : 300
                    let name = "\(kind)_\(transparent ? "transparent" : "opaque")_\(hasData ? "path" : "rounded")"
                    var submit = [[Double](), [Double]()], actual = [0, 0]
                    var observations: [NSKeyValueObservation] = []
                    if kind != .sticker {
                        for variant in 0..<2 {
                            observations.append(try XCTUnwrap(objects[variant].imageLayer()).observe(\.contents, options: [.new]) { _, _ in actual[variant] += 1 })
                        }
                    }
                    defer { observations.forEach { $0.invalidate() } }
                    for batch in 0..<9 {
                        for variant in ((batch % 2 == 0) != reverse ? [0, 1] : [1, 0]) {
                            autoreleasepool {
                                let before = actual[variant]
                                let expected = variant == 0 && transparent && kind != .silhouette && kind != .sticker ? count : 0
                                let start = CACurrentMediaTime()
                                for _ in 0..<count { objects[variant].update(input) }
                                submit[variant].append((CACurrentMediaTime() - start) * 1e6 / Double(count))
                                factory.until { actual[variant] >= before + expected }
                                factory.pump()
                                CATransaction.flush()
                            }
                        }
                    }
                    XCTAssertEqual(shimmerPixels(try XCTUnwrap(objects[0].image())), shimmerPixels(try XCTUnwrap(objects[1].image())))
                    results[name + "_submission"] = submit
                    if kind != .sticker { work[name] = actual }
                }
            }
        }
        TGUIKitBaseline.ShimmerRasterProbe.configure(before: nil)
        TGUIKit.ShimmerRasterProbe.configure(before: nil)
        let changing = [shimmer(.sticker, baseline: true), shimmer(.sticker, baseline: false)]
        for operation in ["changed_sticker_background", "sticker_construction"] {
            var batches = [[Double](), [Double]()]
            let count = operation == "sticker_construction" ? 200 : 1000
            for batch in 0..<9 {
                for variant in ((batch % 2 == 0) != reverse ? [0, 1] : [1, 0]) {
                    autoreleasepool {
                        let start = CACurrentMediaTime()
                        for index in 0..<count {
                            var input = ShimmerInput(); input.background = index % 2 == 0 ? .white : .black
                            if operation == "sticker_construction" { shimmer(.sticker, baseline: variant == 0).update(input) }
                            else { changing[variant].update(input) }
                        }
                        batches[variant].append((CACurrentMediaTime() - start) * 1e6 / Double(count))
                        CATransaction.flush()
                    }
                }
            }
            results[operation] = batches
        }
        try JSONSerialization.data(withJSONObject: ["unit": "microseconds_per_update", "reverse": reverse, "batches": results, "image_deliveries": work, "object_bytes": sizes], options: [.prettyPrinted, .sortedKeys]).write(to: URL(fileURLWithPath: "/tmp/shimmer-render-benchmark.json"))
    }
}
