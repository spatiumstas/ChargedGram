
public enum ShimmerRasterProbe {
    private static let lock = NSLock()
    private static var before: ((CGSize) -> Void)?
    private static var after: ((CGSize) -> Void)?
    public static func configure(before: ((CGSize) -> Void)?, after: ((CGSize) -> Void)? = nil) {
        lock.lock()
        self.before = before
        self.after = after
        lock.unlock()
    }
    static func begin(_ size: CGSize) -> (() -> Void)? {
        lock.lock()
        let before = self.before, after = self.after
        lock.unlock()
        before?(size)
        return after.map { callback in { callback(size) } }
    }
}
