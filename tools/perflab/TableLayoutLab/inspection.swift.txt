
extension TableView {
    public var layoutLabStickyFrames: [NSRect] {
        [stickView?.frame ?? .zero, secondaryStickView?.frame ?? .zero]
    }
    public var layoutLabStickyCacheCounts: [Int] {
        [primaryStickRows?.count ?? -1, secondaryStickRows?.count ?? -1]
    }
}

extension TableView {
    public var layoutLabCachesMatchRows: Bool {
        if let stickClass, let primaryStickRows {
            if primaryStickRows != list.enumerated().compactMap({ $0.element.isKind(of: stickClass) ? $0.offset : nil }) { return false }
        }
        if let secondaryStickClass, let secondaryStickRows {
            if secondaryStickRows != list.enumerated().compactMap({ $0.element.isKind(of: secondaryStickClass) ? $0.offset : nil }) { return false }
        }
        return true
    }
}
