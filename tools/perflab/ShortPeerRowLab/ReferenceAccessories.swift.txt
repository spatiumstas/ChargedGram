        choiceControl?.removeFromSuperview()
        choiceControl = nil

        switch item.type {
        case let .switchable(stateback):
            contextLabel?.removeFromSuperview()
            contextLabel = nil
            contextLabelBg?.removeFromSuperview()
            contextLabelBg = nil
            if switchView == nil {
                switchView = SwitchView()
                containerView.addSubview(switchView!)
            }
            switchView?.stateChanged = item.action
            switchView?.setIsOn(stateback,animated:animated)
            switchView?.isEnabled = item.enabled
        case let .context(stateback), let .nextContext(stateback):
            switchView?.removeFromSuperview()
            switchView = nil
            contextLabelBg?.removeFromSuperview()
            contextLabelBg = nil

            let label = stateback
            if !label.isEmpty {
                if contextLabel == nil {
                    let tv = TextView()
                    tv.userInteractionEnabled = false
                    tv.isSelectable = false
                    containerView.addSubview(tv)
                    contextLabel = tv
                }
                let layout = TextViewLayout(.initialize(string: label, color: item.customTheme?.grayTextColor ?? theme.colors.grayText, font: item.statusStyle.font))
                layout.measure(width: .greatestFiniteMagnitude)
                contextLabel?.update(layout)
                contextLabel?.setFrameSize(layout.layoutSize)
                contextLabel?.backgroundColor = .clear
                contextLabel?.layer?.cornerRadius = 0
                contextLabel?.removeAllHandlers()
                if let badgeAction = item.badgeAction {
                    contextLabel?.scaleOnClick = true
                    contextLabel?.userInteractionEnabled = true
                    contextLabel?.setSingle(handler: { control in
                        badgeAction(control)
                    }, for: .Click)
                } else {
                    contextLabel?.scaleOnClick = false
                    contextLabel?.userInteractionEnabled = false
                }
            } else {
                contextLabel?.removeFromSuperview()
                contextLabel = nil
            }
        case let .contextColored(stateback, color):
            switchView?.removeFromSuperview()
            switchView = nil
            contextLabelBg?.removeFromSuperview()
            contextLabelBg = nil

            let label = stateback
            if !label.isEmpty {
                if contextLabel == nil {
                    let tv = TextView()
                    tv.isSelectable = false
                    containerView.addSubview(tv)
                    contextLabel = tv
                }
                let layout = TextViewLayout(.initialize(string: label, color: color, font: item.statusStyle.font), alignment: .center)
                layout.measure(width: .greatestFiniteMagnitude)
                contextLabel?.update(layout)
                let padding: CGFloat = 5
                contextLabel?.setFrameSize(NSMakeSize(layout.layoutSize.width + padding * 2, layout.layoutSize.height + 2))
                contextLabel?.backgroundColor = color.withAlphaComponent(0.2)
                contextLabel?.layer?.cornerRadius = (layout.layoutSize.height + 2) / 2
                contextLabel?.removeAllHandlers()
                if let badgeAction = item.badgeAction {
                    contextLabel?.scaleOnClick = true
                    contextLabel?.userInteractionEnabled = true
                    contextLabel?.setSingle(handler: { control in
                        badgeAction(control)
                    }, for: .Click)
                } else {
                    contextLabel?.scaleOnClick = false
                    contextLabel?.userInteractionEnabled = false
                }
            } else {
                contextLabel?.removeFromSuperview()
                contextLabel = nil
            }
        case let .selectable(stateback):
            switchView?.removeFromSuperview()
            switchView = nil
            contextLabel?.removeFromSuperview()
            contextLabel = nil
            contextLabelBg?.removeFromSuperview()
            contextLabelBg = nil
            if stateback {
                choiceControl = ImageView()
                choiceControl?.isAccessibilityDecorative = true
                choiceControl?.image = TelegramThemeHooks.shared.image(.iconUsernameAvailability).precomposed(item.customTheme?.accentColor ?? theme.colors.accent)
                choiceControl?.sizeToFit()
                containerView.addSubview(choiceControl!)
            }
        default:
            switchView?.removeFromSuperview()
            switchView = nil
            contextLabel?.removeFromSuperview()
            contextLabel = nil
            contextLabelBg?.removeFromSuperview()
            contextLabelBg = nil
            choiceControl?.removeFromSuperview()
            choiceControl = nil
            break
        }

        if !item.rightActions.actions.isEmpty {
            while self.rightsActions.count > item.rightActions.actions.count {
                self.rightsActions.removeLast().removeFromSuperview()
            }
            while self.rightsActions.count < item.rightActions.actions.count {
                let view = ImageButton()
                containerView.addSubview(view)
                self.rightsActions.append(view)
            }

            for (i, action) in item.rightActions.actions.enumerated() {
                let view = self.rightsActions[i]
                view.set(image: action.icon, for: .Normal)
                view.setAccessibilityLabel(action.accessibilityLabel)
                view.scaleOnClick = true
                view.autohighlight = false
                view.sizeToFit()
                view.setSingle(handler: { [weak item] _ in
                    if let item {
                        item.rightActions.callback(item.peerId, i)
                    }
                }, for: .Click)
            }

        } else {
            while !self.rightsActions.isEmpty {
                self.rightsActions.removeLast().removeFromSuperview()
            }
        }
