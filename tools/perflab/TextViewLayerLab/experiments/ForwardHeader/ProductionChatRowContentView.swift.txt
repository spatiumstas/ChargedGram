import Cocoa
import TGUIKit

final class ChatRowContentView: View {
    private(set) var forwardHeader: ChatTextLayer?
    private var forwardHeaderAccessibility: ChatForwardHeaderAccessibilityElement?
    private var forwardHeaderSubviewIndex = 0

    func updateForwardHeader(_ layout: TextViewLayout?, frame: CGRect, animated: Bool) {
        if let layout {
            let current: ChatTextLayer
            if let existing = forwardHeader {
                current = existing
            } else {
                current = ChatTextLayer()
                current.zPosition = 0
                current.frame = frame
                layer?.addSublayer(current)
                forwardHeader = current
                forwardHeaderSubviewIndex = subviews.count
            }
            current.update(layout)
            current.updateScale(backingScaleFactor)
            needsLayout = true
        } else if let current = forwardHeader {
            performSublayerRemoval(current, animated: animated)
            forwardHeader = nil
            forwardHeaderAccessibility = nil
        }
    }

    func layoutForwardHeader(frame: CGRect, transition: ContainedViewLayoutTransition) {
        if let current = forwardHeader {
            transition.updateFrame(layer: current, frame: frame)
            current.updateDrawLayer()
        }
    }

    func containsForwardHeader(at point: NSPoint) -> Bool {
        guard let forwardHeader, !forwardHeader.isHidden else { return false }
        return NSPointInRect(point, forwardHeader.frame)
    }

    override func viewDidMoveToWindow() {
        super.viewDidMoveToWindow()
        forwardHeader?.updateScale(backingScaleFactor)
        needsLayout = true
    }

    override func didAddSubview(_ subview: NSView) {
        super.didAddSubview(subview)
        if forwardHeader != nil, let index = subviews.firstIndex(of: subview), index < forwardHeaderSubviewIndex {
            forwardHeaderSubviewIndex += 1
        }
        needsLayout = true
    }

    override func willRemoveSubview(_ subview: NSView) {
        if forwardHeader != nil, let index = subviews.firstIndex(of: subview), index < forwardHeaderSubviewIndex {
            forwardHeaderSubviewIndex -= 1
        }
        super.willRemoveSubview(subview)
        needsLayout = true
    }

    override func layout() {
        super.layout()
        guard let layer, let header = forwardHeader, let layers = layer.sublayers,
              let headerIndex = layers.firstIndex(of: header) else { return }
        let index = min(forwardHeaderSubviewIndex, subviews.count)
        if index < subviews.count, let next = subviews[index].layer, let nextIndex = layers.firstIndex(of: next) {
            if headerIndex + 1 != nextIndex {
                layer.insertSublayer(header, below: next)
            }
        } else if let previous = subviews.last?.layer, let previousIndex = layers.firstIndex(of: previous), headerIndex != previousIndex + 1 {
            layer.insertSublayer(header, above: previous)
        }
    }

    override func viewDidChangeBackingProperties() {
        super.viewDidChangeBackingProperties()
        forwardHeader?.updateScale(backingScaleFactor)
    }

    private var headerAccessibilityElement: ChatForwardHeaderAccessibilityElement? {
        guard TGUIKitAccessibility.isEnabled, forwardHeader != nil else {
            return nil
        }
        if forwardHeaderAccessibility == nil {
            forwardHeaderAccessibility = ChatForwardHeaderAccessibilityElement(owner: self)
        }
        return forwardHeaderAccessibility?.isAccessibilityElement() == true ? forwardHeaderAccessibility : nil
    }

    override func accessibilityChildren() -> [Any]? {
        guard TGUIKitAccessibility.isEnabled else { return nil }
        if let element = headerAccessibilityElement {
            let index = min(forwardHeaderSubviewIndex, subviews.count)
            return NSAccessibility.unignoredChildren(from: Array(subviews.prefix(index))) + [element] + NSAccessibility.unignoredChildren(from: Array(subviews.dropFirst(index)))
        }
        return NSAccessibility.unignoredChildren(from: subviews)
    }

    override func accessibilityHitTest(_ point: NSPoint) -> Any? {
        if let element = headerAccessibilityElement, element.accessibilityFrame().contains(point) {
            return element
        }
        return super.accessibilityHitTest(point)
    }
}

private final class ChatForwardHeaderAccessibilityElement: NSAccessibilityElement {
    private weak var owner: ChatRowContentView?

    init(owner: ChatRowContentView) {
        self.owner = owner
        super.init()
    }

    override func isAccessibilityElement() -> Bool {
        guard TGUIKitAccessibility.isEnabled, let owner, !owner.isHiddenOrHasHiddenAncestor,
              let header = owner.forwardHeader, !header.isHidden else {
            return false
        }
        return header.textLayout?.attributedString.string.isEmpty == false
    }

    override func accessibilityParent() -> Any? {
        guard let owner else { return nil }
        return NSAccessibility.unignoredAncestor(of: owner)
    }

    override func accessibilityRole() -> NSAccessibility.Role? { .staticText }
    override func accessibilityLabel() -> String? { nil }
    override func accessibilityValue() -> Any? { owner?.forwardHeader?.textLayout?.attributedString.string }

    override func accessibilityFrame() -> NSRect {
        guard let owner, let window = owner.window, let header = owner.forwardHeader else { return .zero }
        return window.convertToScreen(owner.convert(header.frame, to: nil))
    }
}
