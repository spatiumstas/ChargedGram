import Cocoa
import XCTest
@testable import TGUIKit
@testable import ChatTextRows

extension LayerTests {
    func headerLayout(_ text: String) -> TextViewLayout {
        let value = TextViewLayout(NSAttributedString(string: text, attributes: [.font: NSFont.systemFont(ofSize: 14), .foregroundColor: NSColor.black]), maximumNumberOfLines: 1)
        value.measure(width: 290)
        return value
    }

    func testHeaderMigrationPixelsGeometryAndViews() {
        let old = LegacyHeaderHost(frame: NSRect(x: 0, y: 0, width: 360, height: 80))
        let new = ChatRowContentView(frame: old.frame)
        for value in ["Forwarded message", "مرحبا بالعالم", "転送されたメッセージ", "Öffentliche Bekanntmachung", "Forwarded 👩🏽‍💻", ""] {
            for origin in [NSPoint(x: 12, y: 8), NSPoint(x: 15.5, y: 30.5)] {
                let text = headerLayout(value)
                let item = HeaderItem(forwardHeader: text, defLeftInset: origin.x, forwardHeaderInset: origin)
                let rect = old.forwardHeaderRect(item)
                old.fillForwardHeader(item, animated: false)
                new.updateForwardHeader(text, frame: rect, animated: false)
                ContainedViewLayoutTransition.immediate.updateFrame(view: old.forwardHeader!, frame: rect)
                new.layoutForwardHeader(frame: rect, transition: .immediate)
                XCTAssertEqual(old.forwardHeader!.frame, new.forwardHeader!.frame)
                let before = ViewSnapshot.render(old), after = ViewSnapshot.render(new)
                XCTAssertTrue(before == after, value)
                if !value.isEmpty { XCTAssertGreaterThan(before.pixels(darkerThan: 230), 20) }
                XCTAssertEqual(viewTreeCount(old), 2)
                XCTAssertEqual(viewTreeCount(new), 1)
            }
        }
    }

    func testHeaderAccessibilityValueFrameRebindAndVisibility() {
        let window = NSWindow(contentRect: NSRect(x: 80, y: 100, width: 400, height: 150), styleMask: [.borderless], backing: .buffered, defer: false)
        window.isReleasedWhenClosed = false
        defer { window.close(); TGUIKitAccessibility.isEnabled = true }
        let old = LegacyHeaderHost(frame: NSRect(x: 12, y: 15, width: 360, height: 80))
        let new = ChatRowContentView(frame: old.frame)
        window.contentView!.addSubview(old)
        window.contentView!.addSubview(new)
        for value in ["Forwarded message", "توجيه الرسالة", "New label"] {
            let text = headerLayout(value), origin = NSPoint(x: 15, y: 8)
            let item = HeaderItem(forwardHeader: text, defLeftInset: origin.x, forwardHeaderInset: origin)
            let rect = old.forwardHeaderRect(item)
            old.fillForwardHeader(item, animated: false)
            new.updateForwardHeader(text, frame: rect, animated: false)
            let elements = AccessibilityAudit.elements(new).filter { $0.accessibilityRole() == .staticText }
            XCTAssertEqual(elements.count, 1)
            guard let element = elements.first else { continue }
            XCTAssertEqual(element.accessibilityValue() as? String, old.forwardHeader!.accessibilityValue() as? String)
            XCTAssertEqual(element.accessibilityFrame(), old.forwardHeader!.accessibilityFrame())
            XCTAssertTrue(element.accessibilityParent() as AnyObject? === old.forwardHeader!.accessibilityParent() as AnyObject?)
            XCTAssertTrue(new.accessibilityHitTest(NSPoint(x: element.accessibilityFrame().midX, y: element.accessibilityFrame().midY)) as AnyObject? === element as AnyObject)
            new.isHidden = true
            XCTAssertFalse(element.isAccessibilityElement())
            new.isHidden = false
            XCTAssertTrue(element.isAccessibilityElement())
            TGUIKitAccessibility.isEnabled = false
            XCTAssertNil(new.accessibilityChildren())
            XCTAssertFalse(element.isAccessibilityElement())
            TGUIKitAccessibility.isEnabled = true
        }
        new.updateForwardHeader(nil, frame: .zero, animated: false)
        XCTAssertTrue(AccessibilityAudit.elements(new).isEmpty)
    }

    func testHeaderRemovalHitTestingAndLifetimes() {
        let old = LegacyHeaderHost(frame: NSRect(x: 0, y: 0, width: 360, height: 80))
        let new = ChatRowContentView(frame: old.frame)
        let parent = View(frame: old.frame)
        parent.addSubview(old)
        parent.addSubview(new)
        let text = headerLayout("Forwarded message"), origin = NSPoint(x: 15, y: 8)
        let item = HeaderItem(forwardHeader: text, defLeftInset: origin.x, forwardHeaderInset: origin)
        old.fillForwardHeader(item, animated: false)
        new.updateForwardHeader(text, frame: old.forwardHeaderRect(item), animated: false)
        for y in stride(from: 0, to: 80, by: 4) {
            for x in stride(from: 0, to: 360, by: 4) {
                let point = NSPoint(x: x, y: y)
                let oldHit = old.hitTest(point)
                XCTAssertTrue(oldHit === old || oldHit === old.forwardHeader)
                XCTAssertTrue(new.hitTest(point) === new)
                XCTAssertEqual(oldHit === old.forwardHeader, new.containsForwardHeader(at: point))
            }
        }
        weak var released = new.forwardHeader
        new.updateForwardHeader(nil, frame: .zero, animated: false)
        XCTAssertNil(new.forwardHeader)
        CATransaction.flush()
        XCTAssertNil(released)
        var retainedElement: AnyObject?
        weak var releasedHost: ChatRowContentView?
        autoreleasepool {
            let host = ChatRowContentView(frame: old.frame)
            host.updateForwardHeader(text, frame: old.forwardHeaderRect(item), animated: false)
            retainedElement = host.accessibilityChildren()?.last as AnyObject?
            releasedHost = host
        }
        CATransaction.flush()
        XCTAssertNil(releasedHost)
        XCTAssertNotNil(retainedElement)
        XCTAssertFalse((retainedElement as? NSAccessibilityProtocol)?.isAccessibilityElement() ?? true)
    }

    func testHeaderScaleAndAnimationParameters() {
        let old = LegacyHeaderHost(frame: NSRect(x: 0, y: 0, width: 360, height: 80))
        let new = ChatRowContentView(frame: old.frame)
        let text = headerLayout("Forwarded message"), origin = NSPoint(x: 15, y: 8)
        let item = HeaderItem(forwardHeader: text, defLeftInset: origin.x, forwardHeaderInset: origin)
        old.fillForwardHeader(item, animated: false)
        new.updateForwardHeader(text, frame: old.forwardHeaderRect(item), animated: false)
        let rect = NSRect(x: 20, y: 25, width: text.layoutSize.width, height: text.layoutSize.height)
        let transition = ContainedViewLayoutTransition.animated(duration: 0.3, curve: .easeOut)
        transition.updateFrame(view: old.forwardHeader!, frame: rect)
        new.layoutForwardHeader(frame: rect, transition: transition)
        XCTAssertEqual(old.forwardHeader!.frame, new.forwardHeader!.frame)
        let before = old.forwardHeader!.layer!.animation(forKey: "position") as? CABasicAnimation
        let after = new.forwardHeader!.animation(forKey: "position") as? CABasicAnimation
        XCTAssertNotNil(before); XCTAssertNotNil(after)
        XCTAssertEqual(before?.fromValue as? NSPoint, after?.fromValue as? NSPoint)
        XCTAssertEqual(before?.toValue as? NSPoint, after?.toValue as? NSPoint)
        XCTAssertEqual(before?.duration, after?.duration)
        for scale in [CGFloat(1), 2, 3] {
            System.updateScaleFactor(scale)
            new.viewDidChangeBackingProperties()
            XCTAssertEqual(new.forwardHeader!.sublayers!.first!.contentsScale, new.backingScaleFactor)
        }
    }
}

extension LayerTests {
    func testHeaderSiblingStacking() {
        for above in [false, true] {
            let old = LegacyHeaderHost(frame: NSRect(x: 0, y: 0, width: 360, height: 80))
            let new = ChatRowContentView(frame: old.frame)
            let text = headerLayout("Forwarded message")
            let item = HeaderItem(forwardHeader: text, defLeftInset: 15, forwardHeaderInset: NSPoint(x: 15, y: 8))
            func addSiblings() {
                for host in [old as View, new] {
                    let sibling = View(frame: NSRect(x: 25, y: 8, width: 30, height: 25))
                    sibling.backgroundColor = .red
                    host.addSubview(sibling)
                }
            }
            if !above { addSiblings() }
            old.fillForwardHeader(item, animated: false)
            new.updateForwardHeader(text, frame: old.forwardHeaderRect(item), animated: false)
            if above { addSiblings() }
            let before = ViewSnapshot.render(old), after = ViewSnapshot.render(new)
            print("header ordering", above, old.layer!.sublayers!.map { "\(type(of: $0)) \($0.zPosition) \($0.frame)" }, new.layer!.sublayers!.map { "\(type(of: $0)) \($0.zPosition) \($0.frame)" })
            XCTAssertTrue(before == after, "above=\(above), pixels=\(before.differingPixels(from: after, tolerance: 0))")
        }
    }
}

private final class HeaderEventSink: View {
    var events: [NSEvent.EventType] = []
    override func mouseDown(with event: NSEvent) { events.append(event.type) }
    override func mouseUp(with event: NSEvent) { events.append(event.type) }
    override func mouseDragged(with event: NSEvent) { events.append(event.type) }
    override func rightMouseDown(with event: NSEvent) { events.append(event.type) }
    override func rightMouseUp(with event: NSEvent) { events.append(event.type) }
}

extension LayerTests {
    func testHeaderForwardsMouseEvents() throws {
        let text = headerLayout("Forwarded message"), origin = NSPoint(x: 15, y: 8)
        let item = HeaderItem(forwardHeader: text, defLeftInset: origin.x, forwardHeaderInset: origin)
        let old = LegacyHeaderHost(frame: NSRect(x: 0, y: 0, width: 360, height: 80))
        let new = ChatRowContentView(frame: old.frame)
        old.fillForwardHeader(item, animated: false)
        new.updateForwardHeader(text, frame: old.forwardHeaderRect(item), animated: false)
        var traces: [[NSEvent.EventType]] = []
        for host in [old as View, new] {
            let sink = HeaderEventSink(frame: host.frame)
            sink.addSubview(host)
            for point in [NSPoint(x: 20, y: 12), NSPoint(x: 300, y: 60)] {
                let target = try XCTUnwrap(host.hitTest(point))
                for type in [NSEvent.EventType.leftMouseDown, .leftMouseDragged, .leftMouseUp, .rightMouseDown, .rightMouseUp] {
                    let event = try XCTUnwrap(NSEvent.mouseEvent(with: type, location: point, modifierFlags: [], timestamp: 0, windowNumber: 0, context: nil, eventNumber: 1, clickCount: 1, pressure: 1))
                    switch type {
                    case .leftMouseDown: target.mouseDown(with: event)
                    case .leftMouseUp: target.mouseUp(with: event)
                    case .leftMouseDragged: target.mouseDragged(with: event)
                    case .rightMouseDown: target.rightMouseDown(with: event)
                    case .rightMouseUp: target.rightMouseUp(with: event)
                    default: break
                    }
                }
            }
            traces.append(sink.events)
            host.removeFromSuperview()
        }
        XCTAssertEqual(traces[0].count, 10)
        XCTAssertEqual(traces[0], traces[1])
    }
}
