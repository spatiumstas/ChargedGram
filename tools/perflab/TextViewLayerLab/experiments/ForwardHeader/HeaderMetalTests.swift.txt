import Cocoa
import XCTest
import Metal
@testable import TGUIKit
@testable import ChatTextRows

extension LayerTests {
    func testHeaderAnimatedMetalFrames() throws {
        let device = try XCTUnwrap(MTLCreateSystemDefaultDevice())
        let queue = try XCTUnwrap(device.makeCommandQueue())
        let window = NSWindow(contentRect: NSRect(x: 0, y: 0, width: 360, height: 80), styleMask: [.borderless], backing: .buffered, defer: false)
        window.isReleasedWhenClosed = false
        defer { window.close() }
        for removing in [false, true] {
            var variants: [[Data]] = []
            for candidate in [false, true] {
                let old = LegacyHeaderHost(frame: NSRect(x: 0, y: 0, width: 360, height: 80))
                let new = ChatRowContentView(frame: old.frame)
                let host: View = candidate ? new : old
                window.contentView = host
                let text = headerLayout("Forwarded message 👩🏽‍💻")
                let item = HeaderItem(forwardHeader: text, defLeftInset: 15, forwardHeaderInset: NSPoint(x: 15, y: 8))
                old.fillForwardHeader(item, animated: false)
                new.updateForwardHeader(text, frame: old.forwardHeaderRect(item), animated: false)
                host.layoutSubtreeIfNeeded()
                window.displayIfNeeded()
                func clear(_ layer: CALayer) {
                    layer.removeAllAnimations()
                    layer.sublayers?.forEach(clear)
                }
                clear(host.layer!)
                if removing {
                    if candidate { new.updateForwardHeader(nil, frame: .zero, animated: true) }
                    else { old.fillForwardHeader(HeaderItem(forwardHeader: nil, defLeftInset: 0, forwardHeaderInset: .zero), animated: true) }
                } else {
                    let rect = NSRect(x: 65, y: 45, width: text.layoutSize.width, height: text.layoutSize.height)
                    let transition = ContainedViewLayoutTransition.animated(duration: 4, curve: .easeOut)
                    if candidate { new.layoutForwardHeader(frame: rect, transition: transition) }
                    else { transition.updateFrame(view: old.forwardHeader!, frame: rect) }
                }
                var animations: [(CALayer, String, CAAnimation)] = []
                func collect(_ layer: CALayer) {
                    for key in layer.animationKeys() ?? [] {
                        animations.append((layer, key, layer.animation(forKey: key)!.copy() as! CAAnimation))
                    }
                    layer.sublayers?.forEach(collect)
                }
                collect(host.layer!)
                XCTAssertFalse(animations.isEmpty)
                let renderLayer = host.layer!
                window.contentView = nil
                let root = CALayer()
                root.frame = host.bounds
                root.backgroundColor = NSColor.white.cgColor
                root.addSublayer(renderLayer)
                renderLayer.frame = host.bounds
                let descriptor = MTLTextureDescriptor.texture2DDescriptor(pixelFormat: .bgra8Unorm, width: 360, height: 80, mipmapped: false)
                descriptor.usage = [.renderTarget, .shaderRead]
                descriptor.storageMode = .private
                let texture = try XCTUnwrap(device.makeTexture(descriptor: descriptor))
                let renderer = CARenderer(mtlTexture: texture, options: [kCARendererMetalCommandQueue as String: queue])
                renderer.layer = root
                renderer.bounds = host.bounds
                CATransaction.flush()
                var frames: [Data] = []
                for time in [Double(0.1), 1, 2, 3] {
                    for (layer, key, original) in animations {
                        let animation = original.copy() as! CAAnimation
                        animation.delegate = nil
                        animation.speed = 0
                        animation.beginTime = 0
                        animation.timeOffset = time
                        animation.duration = 4
                        animation.fillMode = .both
                        animation.isRemovedOnCompletion = false
                        layer.add(animation, forKey: key)
                    }
                    CATransaction.flush()
                    renderer.beginFrame(atTime: CACurrentMediaTime(), timeStamp: nil)
                    renderer.addUpdate(renderer.bounds)
                    renderer.render()
                    renderer.endFrame()
                    let buffer = try XCTUnwrap(device.makeBuffer(length: 360 * 80 * 4, options: .storageModeShared))
                    let command = queue.makeCommandBuffer()!
                    let blit = command.makeBlitCommandEncoder()!
                    blit.copy(from: texture, sourceSlice: 0, sourceLevel: 0, sourceOrigin: MTLOrigin(x: 0, y: 0, z: 0), sourceSize: MTLSize(width: 360, height: 80, depth: 1), to: buffer, destinationOffset: 0, destinationBytesPerRow: 1440, destinationBytesPerImage: 115200)
                    blit.endEncoding(); command.commit(); command.waitUntilCompleted()
                    XCTAssertEqual(command.status, .completed)
                    let data = Data(bytes: buffer.contents(), count: 115200)
                    XCTAssertTrue(data.contains { $0 > 0 && $0 < 245 })
                    frames.append(data)
                    if let path = ProcessInfo.processInfo.environment["HEADER_RENDER_DIR"] {
                        let directory = URL(fileURLWithPath: path)
                        try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
                        let image = CGImage(width: 360, height: 80, bitsPerComponent: 8, bitsPerPixel: 32, bytesPerRow: 1440, space: CGColorSpaceCreateDeviceRGB(), bitmapInfo: CGBitmapInfo(rawValue: CGImageAlphaInfo.premultipliedFirst.rawValue | CGBitmapInfo.byteOrder32Little.rawValue), provider: CGDataProvider(data: data as CFData)!, decode: nil, shouldInterpolate: false, intent: .defaultIntent)!
                        try NSBitmapImageRep(cgImage: image).representation(using: .png, properties: [:])!.write(to: directory.appendingPathComponent("\(candidate ? "candidate" : "baseline")-\(removing ? "removal" : "move")-\(time).png"))
                    }
                }
                XCTAssertNotEqual(frames.first, frames.last)
                variants.append(frames)
            }
            for (a, b) in zip(variants[0], variants[1]) { XCTAssertEqual(a, b) }
        }
    }
}
