import hashlib,json,pathlib,subprocess
out=pathlib.Path('tools/perflab/ChatLayerEngineLab/results/forward-name')
hashes=json.load(open('/tmp/chat-forward-inputs.json'))
def verify():
 changed=[p for p,h in hashes.items() if hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()!=h]
 assert not changed,changed
commands={}
jobs=[]
for name,sanitizer in [('asan','address'),('tsan','thread')]:
 cmd=['swift','test','--package-path','tools/perflab/ChatLayerEngineLab','--scratch-path','/tmp/ChatLayerEngineLab-'+name,'--sanitize',sanitizer,'--jobs','4']
 if name=='tsan':cmd+=['--filter','ForwardNameTests|DeferredDispatchTests|DeferredSceneTests|LinkInteractionTests|RowRoutingTests|RowHeaderTests|SceneTests|SenderNameTests|TextReentrancyTests']
 jobs.append((name,cmd))
jobs.append(('tguikit',['swift','test','--package-path','packages/TGUIKit','--scratch-path','/tmp/ChatLayerEngineLab-tguikit','--jobs','4']))
for name,cmd in jobs:
 verify()
 with (out/(name+'.log')).open('w') as log:result=subprocess.run(cmd,stdout=log,stderr=subprocess.STDOUT)
 commands[name]={'command':cmd,'exit_code':result.returncode}
 (out/'validation-commands.json').write_text(json.dumps(commands,indent=2)+'\n')
 print(name,'exit',result.returncode,flush=True)
 verify()
 if result.returncode:raise SystemExit(result.returncode)
print('Verified',len(hashes),'unchanged inputs',flush=True)
