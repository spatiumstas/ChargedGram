import os,json,hashlib,pathlib,subprocess,shutil
out=pathlib.Path('tools/perflab/ChatLayerEngineLab/results/forward-name')
hashes=json.load(open('/tmp/chat-forward-inputs.json'))
status_file=out/'measurement-commands.json'
status=json.load(status_file.open()) if status_file.exists() else {}
def processes():
 return [s for s in subprocess.check_output(['ps','-axo','pid=,comm='],text=True).splitlines() if pathlib.Path(s.strip().split(None,1)[-1]).name in ['swift-test','swift-build','xctest','xcodebuild','swift-frontend']]
def verify():
 changed=[p for p,h in hashes.items() if hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()!=h]
 assert not changed,changed
jobs=[]
for photo in [False,True]:
 for reverse in [False,True]:
  jobs.append(('benchmark-photo-'+str(int(photo))+('-repeat' if reverse else ''),'ForwardNameBenchmarkTests.testForwardBindingCosts',{'CHAT_FORWARD_BENCHMARK':'1','CHAT_FORWARD_PHOTO':str(int(photo)),'CHAT_FORWARD_REVERSE':str(int(reverse))},'/tmp/chat-forward-benchmark.json'))
for batch in range(5):
 for present,photo in [(False,False),(True,False),(True,True)]:
  for mode in (['native','layers'] if batch%2==0 else ['layers','native']):
   jobs.append((f'memory-{batch}-present-{int(present)}-photo-{int(photo)}-{mode}','ForwardNameBenchmarkTests.testRetainedSenderMemory',{'CHAT_FORWARD_MEMORY':mode,'CHAT_FORWARD_PRESENT':str(int(present)),'CHAT_FORWARD_PHOTO':str(int(photo))},'/tmp/chat-forward-memory-'+mode+'.json'))
for name,test,env_vars,result_file in jobs:
 if status.get(name,{}).get('exit_code')==0 and (out/(name+'.json')).exists():continue
 verify()
 before=processes()
 if before and not name.startswith('memory-'):
  print('Deferring measurements for active build/test processes:',before,flush=True)
  raise SystemExit(20)
 cmd=['swift','test','--package-path','tools/perflab/ChatLayerEngineLab','--scratch-path','/tmp/ChatLayerEngineLab-release','-c','release','--skip-build','--filter',test]
 pathlib.Path(result_file).unlink(missing_ok=True)
 with (out/(name+'.log')).open('w') as log:result=subprocess.run(cmd,env=dict(os.environ,**env_vars),stdout=log,stderr=subprocess.STDOUT)
 status[name]={'command':cmd,'environment':env_vars,'before':before,'after':processes(),'exit_code':result.returncode}
 status_file.write_text(json.dumps(status,indent=2)+'\n')
 print(name,'exit',result.returncode,flush=True)
 if result.returncode:raise SystemExit(result.returncode)
 assert pathlib.Path(result_file).exists(),result_file
 shutil.copy(result_file,out/(name+'.json'))
verify()
print('All measurements complete; frozen inputs unchanged',flush=True)
