import Foundation
import CoreFoundation
import QuartzCore

final class Value: NSObject {
    let value: Int
    init(_ value: Int) { self.value = value }
    override var hash: Int { value }
    override func isEqual(_ other: Any?) -> Bool { (other as? Value)?.value == value }
}
let key = NSAttributedString.Key("quote")
let style = NSAttributedString.Key("style")
func forward(_ text: NSAttributedString, _ range: NSRange) -> Any? {
    var result: Any?
    text.enumerateAttribute(key, in: range) { value, _, _ in if let value { result = value } }
    return result
}
func reverse(_ text: NSAttributedString, _ range: NSRange) -> Any? {
    var result: Any?
    text.enumerateAttribute(key, in: range, options: .reverse) { value, _, stop in
        if let value { result = value; stop.pointee = true }
    }
    return result
}
func cfReverse(_ text: NSAttributedString, _ range: NSRange) -> Any? {
    var index = NSMaxRange(range)
    let value = text as CFAttributedString
    while index > range.location {
        var run = CFRange()
        let item = CFAttributedStringGetAttribute(value, index - 1, key.rawValue as CFString, &run)
        if let item { return item }
        index = run.location
    }
    return nil
}
func hybrid(_ text: NSAttributedString, _ range: NSRange) -> Any? {
    guard range.length > 0 else { return nil }
    let string = text as CFAttributedString
    let key = key.rawValue as CFString
    let index = NSMaxRange(range) - 1
    if let value = CFAttributedStringGetAttribute(string, index, key, nil) { return value }
    var missing = CFRange()
    _ = CFAttributedStringGetAttributeAndLongestEffectiveRange(string, index, key, CFRange(location: range.location, length: range.length), &missing)
    guard missing.location > range.location else { return nil }
    return CFAttributedStringGetAttribute(string, missing.location - 1, key, nil)
}
var failures = [0,0,0]
var comparisons = 0
for sample in 0..<2048 {
    let text = NSMutableAttributedString(string: "ab👩🏽‍💻 مرحبا zzz")
    for index in 0..<text.length {
        text.addAttribute(style, value: index % 3, range: NSRange(location: index, length: 1))
        if (sample >> (index % 11)) & 1 != 0 {
            text.addAttribute(key, value: Value(sample % 2 == 0 ? (sample + index) % 3 : 1), range: NSRange(location: index, length: 1))
        }
    }
    for start in [0, 1, text.length / 2] {
        let range = NSRange(location: start, length: text.length - start - sample % 2)
        let expected = forward(text,range) as AnyObject?
        for (mode,op) in [reverse,cfReverse,hybrid].enumerated() {
            let actual = op(text,range) as AnyObject?
            if actual !== expected { failures[mode] += 1 }
        }
        comparisons += 1
    }
}
print("identity",comparisons,failures)
var results: [String: [Double]] = [:]
var sink = 0
for runs in [1,16,128] {
 for variant in 0..<4 {
    let text = NSMutableAttributedString(string: String(repeating: "abcd", count: runs))
    for run in 0..<runs { text.addAttribute(style, value: run, range: NSRange(location: run * 4, length: 4)) }
    if variant == 1 { text.addAttribute(key, value: Value(1), range: NSRange(location: 0, length: 1)) }
    if variant == 2 { text.addAttribute(key, value: Value(1), range: NSRange(location: text.length-1, length: 1)) }
    if variant == 3 { text.addAttribute(key, value: Value(1), range: NSRange(location: 0, length: text.length)) }
    for (mode,op) in [forward,reverse,cfReverse,hybrid].enumerated() {
        for _ in 0..<5 {
            let start=CACurrentMediaTime()
            for _ in 0..<10000 { if op(text,NSRange(location:0,length:text.length)) != nil { sink += 1 } }
            results["\(runs)-\(variant)-\(mode)",default:[]].append((CACurrentMediaTime()-start)*1e9/10000)
        }
    }
 }
}
try JSONSerialization.data(withJSONObject: ["nanoseconds":results,"sink":sink,"identity_comparisons":comparisons,"identity_failures":failures],options:[.prettyPrinted,.sortedKeys]).write(to:URL(fileURLWithPath:"/tmp/attribute-scan-experiment.json"))
