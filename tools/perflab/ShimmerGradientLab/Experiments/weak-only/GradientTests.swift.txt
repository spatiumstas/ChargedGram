import Cocoa
import XCTest
import class TGUIKit.GradientLayerHarness
import class TGUIKitBaseline.GradientLayerHarness
import class TGUIKit.ShimmerEffectForegroundView
import class TGUIKitBaseline.ShimmerEffectForegroundView
import enum TGUIKit.GradientCacheInspection
import enum TGUIKitBaseline.GradientCacheInspection
import struct TGUIKit.DeviceGraphicsContextSettings
import struct TGUIKitBaseline.DeviceGraphicsContextSettings

struct GradientInput {
    var background: NSColor = .clear
    var foreground = NSColor(calibratedRed: 0.4, green: 0.7, blue: 0.8, alpha: 0.35)
    var length: CGFloat? = nil
    var duration: Double? = nil
    var horizontal = true
}

struct GradientHarness {
    let owner: AnyObject
    let object: AnyObject
    let update: (GradientInput) -> Void
    let image: () -> CGImage?
}

func gradient(layer: Bool, baseline: Bool) -> GradientHarness {
    if layer {
        if baseline {
            let value = TGUIKitBaseline.GradientLayerHarness()
            return GradientHarness(owner: value, object: value.object, update: { value.update(background: $0.background, foreground: $0.foreground) }, image: { value.image })
        } else {
            let value = TGUIKit.GradientLayerHarness()
            return GradientHarness(owner: value, object: value.object, update: { value.update(background: $0.background, foreground: $0.foreground) }, image: { value.image })
        }
    } else {
        if baseline {
            let value = TGUIKitBaseline.ShimmerEffectForegroundView()
            return GradientHarness(owner: value, object: value, update: { value.update(backgroundColor: $0.background, foregroundColor: $0.foreground, gradientSize: $0.length, duration: $0.duration, horizontal: $0.horizontal) }, image: { value.imageForGradientLab })
        } else {
            let value = TGUIKit.ShimmerEffectForegroundView()
            return GradientHarness(owner: value, object: value, update: { value.update(backgroundColor: $0.background, foregroundColor: $0.foreground, gradientSize: $0.length, duration: $0.duration, horizontal: $0.horizontal) }, image: { value.imageForGradientLab })
        }
    }
}

func gradientPixels(_ image: CGImage) -> [UInt8] {
    var pixels = [UInt8](repeating: 0, count: image.width * image.height * 4)
    pixels.withUnsafeMutableBytes { bytes in
        let context = CGContext(data: bytes.baseAddress, width: image.width, height: image.height, bitsPerComponent: 8, bytesPerRow: image.width * 4, space: CGColorSpace(name: CGColorSpace.sRGB)!, bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue)!
        context.setBlendMode(.copy)
        context.draw(image, in: CGRect(x: 0, y: 0, width: image.width, height: image.height))
    }
    return pixels
}

final class GradientTests: XCTestCase {
    override func setUp() {
        MeasureBootstrap.install()
        TGUIKit.GradientCacheInspection.reset(); TGUIKitBaseline.GradientCacheInspection.reset()
        TGUIKit.DeviceGraphicsContextSettings.install(nil); TGUIKitBaseline.DeviceGraphicsContextSettings.install(nil)
    }
    override func tearDown() {
        TGUIKit.GradientCacheInspection.reset(); TGUIKitBaseline.GradientCacheInspection.reset()
        TGUIKit.DeviceGraphicsContextSettings.install(nil); TGUIKitBaseline.DeviceGraphicsContextSettings.install(nil)
    }
    func flush() {
        CATransaction.flush()
        let end = CACurrentMediaTime() + 0.03
        while CACurrentMediaTime() < end { _ = RunLoop.main.run(mode: .default, before: Date(timeIntervalSinceNow: 0.001)) }
    }

    func testFreshAndChangedGradientPixelsMatch() throws {
        let foregrounds: [NSColor] = [.white, .black, .clear, NSColor(calibratedRed: 0.123, green: 0.567, blue: 0.891, alpha: 0.35), NSColor(displayP3Red: 1, green: 0.2, blue: 0.4, alpha: 0.75)]
        let backgrounds: [NSColor] = [.clear, .white, .black, NSColor(calibratedRed: 0.4, green: 0.2, blue: 0.7, alpha: 0.4)]
        var pairs = 0
        for layer in [false, true] {
            let reused = gradient(layer: layer, baseline: false)
            for foreground in foregrounds {
                for background in backgrounds {
                    for horizontal in layer ? [true] : [false, true] {
                        let lengths: [CGFloat?] = layer ? [nil] : [nil, 0, 1, 17.5, 320, 511]
                        for length in lengths {
                            var input = GradientInput(); input.foreground = foreground; input.background = background; input.horizontal = horizontal; input.length = length
                            let original = gradient(layer: layer, baseline: true)
                            original.update(input); reused.update(input)
                            XCTAssertEqual(original.image() == nil, reused.image() == nil)
                            if let a = original.image(), let b = reused.image() {
                                XCTAssertEqual(a.width, b.width); XCTAssertEqual(a.height, b.height)
                                XCTAssertEqual(gradientPixels(a), gradientPixels(b), "layer=\(layer) horizontal=\(horizontal) length=\(String(describing: length))")
                                pairs += 1
                            }
                        }
                    }
                }
            }
        }
        XCTAssertEqual(pairs, 220)
    }

    func testLiveViewsShareImagesAndReleaseThem() throws {
        for baseline in [true, false] {
            weak var weakImage: CGImage?
            var uniqueCount = 0, bytes = 0
            autoreleasepool {
                var values: [GradientHarness] = (0..<32).map { _ in gradient(layer: false, baseline: baseline) }
                for value in values { value.update(GradientInput()) }
                let images = values.compactMap { $0.image() }
                uniqueCount = Set(images.map(ObjectIdentifier.init)).count
                var seen = Set<ObjectIdentifier>()
                for image in images where seen.insert(ObjectIdentifier(image)).inserted { bytes += image.bytesPerRow * image.height }
                weakImage = images.first
                values.removeAll()
                CATransaction.flush()
            }
            flush()
            XCTAssertEqual(uniqueCount, baseline ? 32 : 1)
            XCTAssertEqual(bytes, (baseline ? 32 : 1) * 320 * 16 * 4)
            XCTAssertNil(weakImage)
        }
        XCTAssertEqual(TGUIKit.GradientCacheInspection.liveImages, 0)
        XCTAssertEqual(TGUIKit.GradientCacheInspection.pixelBytes, 0)
    }

    func testUnusedLayerImagesAreNotRetainedAndMetadataIsBounded() {
        let window = NSWindow(contentRect: CGRect(x: 0, y: 0, width: 100, height: 100), styleMask: [.borderless], backing: .buffered, defer: false)
        window.isReleasedWhenClosed = false
        let host = NSView(frame: window.contentView!.bounds); host.wantsLayer = true; window.contentView = host
        defer { window.close() }
        for baseline in [true, false] {
            weak var owner: AnyObject?
            weak var image: CGImage?
            autoreleasepool {
                for index in 0..<256 {
                    let value = gradient(layer: true, baseline: baseline)
                    var input = GradientInput(); input.foreground = NSColor(deviceRed: CGFloat(index % 16) / 16, green: CGFloat(index / 16) / 16, blue: 0.8, alpha: 0.35)
                    let layer = value.object as! CALayer
                    host.layer!.addSublayer(layer)
                    value.update(input)
                    layer.removeFromSuperlayer()
                    owner = value.object; image = value.image()
                }
                CATransaction.flush()
            }
            autoreleasepool { flush() }
            XCTAssertNil(owner)
            if !baseline { XCTAssertNil(image) }
            if baseline {
                XCTAssertEqual(TGUIKitBaseline.GradientCacheInspection.liveImages, 256)
                XCTAssertEqual(TGUIKitBaseline.GradientCacheInspection.pixelBytes, 256 * 320 * 16 * 4)
            } else {
                XCTAssertLessThanOrEqual(TGUIKit.GradientCacheInspection.metadataCount, 32)
                XCTAssertEqual(TGUIKit.GradientCacheInspection.liveImages, 0)
                XCTAssertEqual(TGUIKit.GradientCacheInspection.pixelBytes, 0)
            }
        }
    }

    func testResolvedDynamicColorsInvalidateExistingGradients() throws {
        if #available(macOS 11, *) {
            for layer in [false, true] {
                var dark = false
                let background = NSColor(name: nil, dynamicProvider: { _ in dark ? .black : .white })
                let foreground = NSColor(name: nil, dynamicProvider: { _ in dark ? .white : .black })
                var input = GradientInput(); input.background = background; input.foreground = foreground
                let old = gradient(layer: layer, baseline: true), new = gradient(layer: layer, baseline: false)
                old.update(input); new.update(input)
                let before = gradientPixels(try XCTUnwrap(old.image()))
                dark = true
                old.update(input); new.update(input)
                XCTAssertEqual(before, gradientPixels(try XCTUnwrap(old.image())))
                XCTAssertNotEqual(before, gradientPixels(try XCTUnwrap(new.image())))
                input.background = .black; input.foreground = .white
                TGUIKitBaseline.GradientCacheInspection.reset()
                let fresh = gradient(layer: layer, baseline: true); fresh.update(input)
                XCTAssertEqual(gradientPixels(try XCTUnwrap(fresh.image())), gradientPixels(try XCTUnwrap(new.image())))
                TGUIKitBaseline.GradientCacheInspection.reset()
            }
        }
    }

    func testCacheEvictionAndDurationChangesKeepRendering() throws {
        var held: [GradientHarness] = []
        for index in 0..<100 {
            let value = gradient(layer: false, baseline: false)
            var input = GradientInput(); input.length = CGFloat(32 + index)
            value.update(input); held.append(value)
            XCTAssertLessThanOrEqual(TGUIKit.GradientCacheInspection.metadataCount, 32)
        }
        let value = try XCTUnwrap(held.last), image = try XCTUnwrap(value.image())
        var input = GradientInput(); input.length = 131; input.duration = 2
        value.update(input)
        XCTAssertTrue(value.image() === image)
        for index in 0..<100 {
            var input = GradientInput(); input.length = CGFloat(32 + index)
            let old = gradient(layer: false, baseline: true), new = gradient(layer: false, baseline: false)
            old.update(input); new.update(input)
            XCTAssertEqual(gradientPixels(try XCTUnwrap(old.image())), gradientPixels(try XCTUnwrap(new.image())))
        }
    }
}

private final class ConcurrentGradientImages {
    private let lock = NSLock()
    private var storage: [Int: [CGImage]] = [:]
    func add(_ image: CGImage, key: Int) { lock.lock(); storage[key, default: []].append(image); lock.unlock() }
    var values: [Int: [CGImage]] { lock.lock(); defer { lock.unlock() }; return storage }
}

extension GradientTests {
    func testConcurrentSharingUsesConsistentLiveImages() throws {
        let images = ConcurrentGradientImages()
        let input = GradientInput(), colorSpace = CGColorSpaceCreateDeviceRGB()
        DispatchQueue.concurrentPerform(iterations: 512) { index in
            let key = index % 16
            let image = TGUIKit.GradientCacheInspection.image(background: input.background, foreground: input.foreground, size: CGSize(width: 64 + key, height: 16), horizontal: true, colorSpace: colorSpace)!
            images.add(image, key: key)
        }
        let values = images.values
        XCTAssertEqual(values.count, 16)
        for (key, group) in values {
            XCTAssertEqual(group.count, 32)
            XCTAssertEqual(Set(group.map(ObjectIdentifier.init)).count, 1)
            var input = GradientInput(); input.length = CGFloat(64 + key)
            let reference = gradient(layer: false, baseline: true); reference.update(input)
            XCTAssertEqual(gradientPixels(try XCTUnwrap(group.first)), gradientPixels(try XCTUnwrap(reference.image())))
        }
        XCTAssertEqual(TGUIKit.GradientCacheInspection.liveImages, 16)
    }

    func testGraphicsSettingsChangesDoNotReuseOldBitmapFormats() throws {
        var held: [GradientHarness] = []
        for spaceName in [CGColorSpace.sRGB, CGColorSpace.displayP3, CGColorSpace.sRGB] {
            let context = try XCTUnwrap(CGContext(data: nil, width: 1, height: 1, bitsPerComponent: 8, bytesPerRow: 32, space: CGColorSpace(name: spaceName)!, bitmapInfo: CGBitmapInfo.byteOrder32Little.rawValue | CGImageAlphaInfo.premultipliedFirst.rawValue))
            TGUIKit.DeviceGraphicsContextSettings.install(context); TGUIKitBaseline.DeviceGraphicsContextSettings.install(context)
            TGUIKitBaseline.GradientCacheInspection.reset()
            for layer in [false, true] {
                let old = gradient(layer: layer, baseline: true), new = gradient(layer: layer, baseline: false)
                old.update(GradientInput()); new.update(GradientInput())
                let reference = try XCTUnwrap(old.image()), result = try XCTUnwrap(new.image())
                XCTAssertTrue(CFEqual(reference.colorSpace!, result.colorSpace!))
                XCTAssertEqual(gradientPixels(reference), gradientPixels(result))
                if let previous = held.last?.image() { XCTAssertFalse(previous === result) }
                held.append(new)
            }
        }
    }

    func testFramesAndAnimationPropertiesMatch() throws {
        let old = TGUIKitBaseline.ShimmerEffectForegroundView()
        let new = TGUIKit.ShimmerEffectForegroundView()
        old.frame = CGRect(x: 0, y: 0, width: 180, height: 100)
        new.frame = CGRect(x: 0, y: 100, width: 180, height: 100)
        let window = NSWindow(contentRect: CGRect(x: 0, y: 0, width: 300, height: 300), styleMask: [.borderless], backing: .buffered, defer: false)
        window.isReleasedWhenClosed = false
        let host = NSView(frame: window.contentView!.bounds); window.contentView = host
        host.addSubview(old); host.addSubview(new)
        defer { window.close() }
        for index in 0..<12 {
            let horizontal = index % 2 == 0, length: CGFloat? = index % 3 == 0 ? nil : CGFloat(80 + index)
            old.update(backgroundColor: .clear, foregroundColor: .white, gradientSize: length, duration: 1.2, horizontal: horizontal)
            new.update(backgroundColor: .clear, foregroundColor: .white, gradientSize: length, duration: 1.2, horizontal: horizontal)
            let rect = CGRect(x: index * 2, y: index * 3, width: 100 + index, height: 70 + index), size = CGSize(width: 200 + index, height: 150 + index)
            old.updateAbsoluteRect(rect, within: size); new.updateAbsoluteRect(rect, within: size)
            XCTAssertEqual(old.imageFrameForGradientLab, new.imageFrameForGradientLab)
            XCTAssertEqual(old.containerFrameForGradientLab, new.containerFrameForGradientLab)
            XCTAssertEqual(old.imageLayerForGradientLab.contentsGravity, new.imageLayerForGradientLab.contentsGravity)
            XCTAssertEqual(old.imageLayerForGradientLab.masksToBounds, new.imageLayerForGradientLab.masksToBounds)
            let a = try XCTUnwrap(old.imageLayerForGradientLab.animation(forKey: "shimmer") as? CABasicAnimation)
            let b = try XCTUnwrap(new.imageLayerForGradientLab.animation(forKey: "shimmer") as? CABasicAnimation)
            XCTAssertEqual(a.keyPath, b.keyPath); XCTAssertEqual(a.duration, b.duration); XCTAssertEqual(a.repeatCount, b.repeatCount)
            XCTAssertEqual(a.fromValue as? NSNumber, b.fromValue as? NSNumber); XCTAssertEqual(a.toValue as? NSNumber, b.toValue as? NSNumber)
        }
        old.removeFromSuperview(); new.removeFromSuperview()
        XCTAssertNil(new.imageLayerForGradientLab.animation(forKey: "shimmer"))
        XCTAssertNil(old.imageLayerForGradientLab.animation(forKey: "shimmer"))
    }
}

extension GradientTests {
    func testSubByteColorsDoNotAliasAndSquareDirectionsStayDistinct() throws {
        let old = gradient(layer: true, baseline: true), new = gradient(layer: true, baseline: false)
        var input = GradientInput(); input.foreground = NSColor(deviceRed: 0.2, green: 0.4, blue: 0.8, alpha: 0.3500)
        old.update(input); new.update(input)
        let original = try XCTUnwrap(old.image()), previous = try XCTUnwrap(new.image())
        input.foreground = NSColor(deviceRed: 0.2, green: 0.4, blue: 0.8, alpha: 0.3510)
        old.update(input); new.update(input)
        XCTAssertTrue(old.image() === original)
        XCTAssertFalse(new.image() === previous)
        TGUIKitBaseline.GradientCacheInspection.reset()
        let fresh = gradient(layer: true, baseline: true); fresh.update(input)
        XCTAssertTrue(gradientPixels(try XCTUnwrap(fresh.image())) == gradientPixels(try XCTUnwrap(new.image())))
        XCTAssertFalse(gradientPixels(previous) == gradientPixels(try XCTUnwrap(new.image())))
        let horizontal = gradient(layer: false, baseline: false), vertical = gradient(layer: false, baseline: false)
        input.length = 16; input.horizontal = true; horizontal.update(input)
        input.horizontal = false; vertical.update(input)
        XCTAssertFalse(horizontal.image() === vertical.image())
        XCTAssertFalse(gradientPixels(try XCTUnwrap(horizontal.image())) == gradientPixels(try XCTUnwrap(vertical.image())))
    }
