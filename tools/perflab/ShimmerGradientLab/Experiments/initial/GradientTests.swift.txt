import Cocoa
import XCTest
import class TGUIKit.GradientLayerHarness
import class TGUIKitBaseline.GradientLayerHarness
import class TGUIKit.ShimmerEffectForegroundView
import class TGUIKitBaseline.ShimmerEffectForegroundView
import enum TGUIKit.GradientCacheInspection
import enum TGUIKitBaseline.GradientCacheInspection
import struct TGUIKit.DeviceGraphicsContextSettings
import struct TGUIKitBaseline.DeviceGraphicsContextSettings

struct GradientInput {
    var background: NSColor = .clear
    var foreground = NSColor(calibratedRed: 0.4, green: 0.7, blue: 0.8, alpha: 0.35)
    var length: CGFloat? = nil
    var duration: Double? = nil
    var horizontal = true
}

struct GradientHarness {
    let owner: AnyObject
    let object: AnyObject
    let update: (GradientInput) -> Void
    let image: () -> CGImage?
}

func gradient(layer: Bool, baseline: Bool) -> GradientHarness {
    if layer {
        if baseline {
            let value = TGUIKitBaseline.GradientLayerHarness()
            return GradientHarness(owner: value, object: value.object, update: { value.update(background: $0.background, foreground: $0.foreground) }, image: { value.image })
        } else {
            let value = TGUIKit.GradientLayerHarness()
            return GradientHarness(owner: value, object: value.object, update: { value.update(background: $0.background, foreground: $0.foreground) }, image: { value.image })
        }
    } else {
        if baseline {
            let value = TGUIKitBaseline.ShimmerEffectForegroundView()
            return GradientHarness(owner: value, object: value, update: { value.update(backgroundColor: $0.background, foregroundColor: $0.foreground, gradientSize: $0.length, duration: $0.duration, horizontal: $0.horizontal) }, image: { value.imageForGradientLab })
        } else {
            let value = TGUIKit.ShimmerEffectForegroundView()
            return GradientHarness(owner: value, object: value, update: { value.update(backgroundColor: $0.background, foregroundColor: $0.foreground, gradientSize: $0.length, duration: $0.duration, horizontal: $0.horizontal) }, image: { value.imageForGradientLab })
        }
    }
}

func gradientPixels(_ image: CGImage) -> [UInt8] {
    var pixels = [UInt8](repeating: 0, count: image.width * image.height * 4)
    pixels.withUnsafeMutableBytes { bytes in
        let context = CGContext(data: bytes.baseAddress, width: image.width, height: image.height, bitsPerComponent: 8, bytesPerRow: image.width * 4, space: CGColorSpace(name: CGColorSpace.sRGB)!, bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue)!
        context.setBlendMode(.copy)
        context.draw(image, in: CGRect(x: 0, y: 0, width: image.width, height: image.height))
    }
    return pixels
}

final class GradientTests: XCTestCase {
    override func setUp() {
        MeasureBootstrap.install()
        TGUIKit.GradientCacheInspection.reset(); TGUIKitBaseline.GradientCacheInspection.reset()
        TGUIKit.DeviceGraphicsContextSettings.install(nil); TGUIKitBaseline.DeviceGraphicsContextSettings.install(nil)
    }
    override func tearDown() {
        TGUIKit.GradientCacheInspection.reset(); TGUIKitBaseline.GradientCacheInspection.reset()
        TGUIKit.DeviceGraphicsContextSettings.install(nil); TGUIKitBaseline.DeviceGraphicsContextSettings.install(nil)
    }
    func flush() {
        CATransaction.flush()
        let end = CACurrentMediaTime() + 0.03
        while CACurrentMediaTime() < end { _ = RunLoop.main.run(mode: .default, before: Date(timeIntervalSinceNow: 0.001)) }
    }

    func testFreshAndChangedGradientPixelsMatch() throws {
        let foregrounds: [NSColor] = [.white, .black, .clear, NSColor(calibratedRed: 0.123, green: 0.567, blue: 0.891, alpha: 0.35), NSColor(displayP3Red: 1, green: 0.2, blue: 0.4, alpha: 0.75)]
        let backgrounds: [NSColor] = [.clear, .white, .black, NSColor(calibratedRed: 0.4, green: 0.2, blue: 0.7, alpha: 0.4)]
        var pairs = 0
        for layer in [false, true] {
            let reused = gradient(layer: layer, baseline: false)
            for foreground in foregrounds {
                for background in backgrounds {
                    for horizontal in layer ? [true] : [false, true] {
                        let lengths: [CGFloat?] = layer ? [nil] : [nil, 0, 1, 17.5, 320, 511]
                        for length in lengths {
                            var input = GradientInput(); input.foreground = foreground; input.background = background; input.horizontal = horizontal; input.length = length
                            let original = gradient(layer: layer, baseline: true)
                            original.update(input); reused.update(input)
                            XCTAssertEqual(original.image() == nil, reused.image() == nil)
                            if let a = original.image(), let b = reused.image() {
                                XCTAssertEqual(a.width, b.width); XCTAssertEqual(a.height, b.height)
                                XCTAssertEqual(gradientPixels(a), gradientPixels(b), "layer=\(layer) horizontal=\(horizontal) length=\(String(describing: length))")
                                pairs += 1
                            }
                        }
                    }
                }
            }
        }
        XCTAssertEqual(pairs, 220)
    }

    func testLiveViewsShareImagesAndReleaseThem() throws {
        for baseline in [true, false] {
            weak var weakImage: CGImage?
            var uniqueCount = 0, bytes = 0
            autoreleasepool {
                var values: [GradientHarness] = (0..<32).map { _ in gradient(layer: false, baseline: baseline) }
                for value in values { value.update(GradientInput()) }
                let images = values.compactMap { $0.image() }
                uniqueCount = Set(images.map(ObjectIdentifier.init)).count
                var seen = Set<ObjectIdentifier>()
                for image in images where seen.insert(ObjectIdentifier(image)).inserted { bytes += image.bytesPerRow * image.height }
                weakImage = images.first
                values.removeAll()
                CATransaction.flush()
            }
            flush()
            XCTAssertEqual(uniqueCount, baseline ? 32 : 1)
            XCTAssertEqual(bytes, (baseline ? 32 : 1) * 320 * 16 * 4)
            XCTAssertNil(weakImage)
        }
        XCTAssertEqual(TGUIKit.GradientCacheInspection.liveImages, 0)
        XCTAssertEqual(TGUIKit.GradientCacheInspection.pixelBytes, 0)
    }

    func testUnusedLayerImagesAreNotRetainedAndMetadataIsBounded() {
        for baseline in [true, false] {
            autoreleasepool {
                for index in 0..<256 {
                    let value = gradient(layer: true, baseline: baseline)
                    var input = GradientInput(); input.foreground = NSColor(calibratedRed: CGFloat(index) / 256, green: 0.3, blue: 0.8, alpha: 0.35)
                    value.update(input)
                }
                CATransaction.flush()
            }
            flush()
            if baseline {
                XCTAssertEqual(TGUIKitBaseline.GradientCacheInspection.liveImages, 256)
                XCTAssertEqual(TGUIKitBaseline.GradientCacheInspection.pixelBytes, 256 * 320 * 16 * 4)
            } else {
                XCTAssertLessThanOrEqual(TGUIKit.GradientCacheInspection.metadataCount, 32)
                XCTAssertEqual(TGUIKit.GradientCacheInspection.liveImages, 0)
                XCTAssertEqual(TGUIKit.GradientCacheInspection.pixelBytes, 0)
            }
        }
    }

    func testResolvedDynamicColorsInvalidateExistingGradients() throws {
        if #available(macOS 11, *) {
            for layer in [false, true] {
                var dark = false
                let background = NSColor(name: nil, dynamicProvider: { _ in dark ? .black : .white })
                let foreground = NSColor(name: nil, dynamicProvider: { _ in dark ? .white : .black })
                var input = GradientInput(); input.background = background; input.foreground = foreground
                let old = gradient(layer: layer, baseline: true), new = gradient(layer: layer, baseline: false)
                old.update(input); new.update(input)
                let before = gradientPixels(try XCTUnwrap(old.image()))
                dark = true
                old.update(input); new.update(input)
                XCTAssertEqual(before, gradientPixels(try XCTUnwrap(old.image())))
                XCTAssertNotEqual(before, gradientPixels(try XCTUnwrap(new.image())))
                input.background = .black; input.foreground = .white
                let fresh = gradient(layer: layer, baseline: true); fresh.update(input)
                XCTAssertEqual(gradientPixels(try XCTUnwrap(fresh.image())), gradientPixels(try XCTUnwrap(new.image())))
                TGUIKitBaseline.GradientCacheInspection.reset()
            }
        }
    }

    func testCacheEvictionAndDurationChangesKeepRendering() throws {
        var held: [GradientHarness] = []
        for index in 0..<100 {
            let value = gradient(layer: false, baseline: false)
            var input = GradientInput(); input.length = CGFloat(32 + index)
            value.update(input); held.append(value)
            XCTAssertLessThanOrEqual(TGUIKit.GradientCacheInspection.metadataCount, 32)
        }
        let value = try XCTUnwrap(held.last), image = try XCTUnwrap(value.image())
        var input = GradientInput(); input.length = 131; input.duration = 2
        value.update(input)
        XCTAssertTrue(value.image() === image)
        for index in 0..<100 {
            var input = GradientInput(); input.length = CGFloat(32 + index)
            let old = gradient(layer: false, baseline: true), new = gradient(layer: false, baseline: false)
            old.update(input); new.update(input)
            XCTAssertEqual(gradientPixels(try XCTUnwrap(old.image())), gradientPixels(try XCTUnwrap(new.image())))
        }
    }
}
