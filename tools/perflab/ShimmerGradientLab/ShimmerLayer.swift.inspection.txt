
private extension ShimmerEffectForegroundLayer {
    var imageForGradientLab: CGImage? { imageView.contents as! CGImage? }
}

public final class GradientLayerHarness {
    private let value = ShimmerEffectForegroundLayer()
    public init() {}
    public var object: AnyObject { value }
    public var layer: CALayer { value }
    public var image: CGImage? { value.imageForGradientLab }
    public func update(background: NSColor, foreground: NSColor) {
        value.update(backgroundColor: background, foregroundColor: foreground)
    }
}
