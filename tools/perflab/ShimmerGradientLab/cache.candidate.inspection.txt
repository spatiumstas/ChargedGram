
private extension ShimmerGradientCache {
    func resetForLab() {
        lock.lock()
        defer { lock.unlock() }
        entries.removeAll()
        recent.removeAll()
        retainedBytes = 0
        order = Array(repeating: nil, count: 32)
        next = 0
        settings = nil
    }
    var retainedBytesForLab: Int {
        lock.lock()
        defer { lock.unlock() }
        return retainedBytes
    }
    var countsForLab: (Int, Int, Int) {
        lock.lock()
        defer { lock.unlock() }
        let images = entries.values.compactMap { $0.image }
        return (entries.count, images.count, images.reduce(0) { $0 + $1.bytesPerRow * $1.height })
    }
}

public enum GradientCacheInspection {
    public static func reset() { ShimmerGradientCache.shared.resetForLab() }
    public static var metadataCount: Int { ShimmerGradientCache.shared.countsForLab.0 }
    public static var liveImages: Int { ShimmerGradientCache.shared.countsForLab.1 }
    public static var retainedPixelBytes: Int { ShimmerGradientCache.shared.retainedBytesForLab }
    public static var pixelBytes: Int { ShimmerGradientCache.shared.countsForLab.2 }
    public static func image(background: NSColor, foreground: NSColor, size: CGSize, horizontal: Bool, colorSpace: CGColorSpace) -> CGImage? {
        shimmerGradientImage(backgroundColor: background, foregroundColor: foreground, size: size, horizontal: horizontal, gradientColorSpace: colorSpace)
    }
}
