extension BlurEdgeMaskCache {
    fileprivate var countForLab: Int {
        lock.lock()
        defer { lock.unlock() }
        return entries.count
    }
}

    public static var count: Int { BlurEdgeMaskCache.shared.countForLab }
    public static func image(size: CGSize, padding: CGFloat, bottom: Bool, generate: () -> CIImage?) -> CIImage? {
        BlurEdgeMaskCache.shared.image(size: size, padding: padding, bottom: bottom, generate: generate)
    }
}
