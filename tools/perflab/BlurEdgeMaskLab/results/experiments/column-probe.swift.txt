import Cocoa
import CoreImage

class OriginalMask {
    enum Edge { case top, bottom }
    var currentEdge: Edge = .bottom
    var padding: CGFloat = 0
    var bitmapBytes = 0
    func verticalGradient(size: CGSize) -> CIImage? {
        guard let ctx = CGContext(
            data: nil,
            width: Int(round(size.width)),
            height: Int(round(size.height)),
            bitsPerComponent: 8,
            bytesPerRow: 0,
            space: CGColorSpaceCreateDeviceGray(),
            bitmapInfo: CGImageAlphaInfo.none.rawValue
        ) else { return nil }

        let colors: [NSColor]
        switch currentEdge {
        case .bottom:
            colors = [.white, .black]
        case .top:
            colors = [.black, .white]
        }
        let cgcolors = colors.map { $0.cgColor } as CFArray

        guard let gradient = CGGradient(
            colorsSpace: CGColorSpaceCreateDeviceGray(),
            colors: cgcolors,
            locations: [0.0, 1.0]
        ) else { return nil }

        ctx.drawLinearGradient(
            gradient,
            start: CGPoint(x: 0, y: -padding),
            end: CGPoint(x: 0, y: size.height + padding),
            options: []
        )

        bitmapBytes = ctx.bytesPerRow * ctx.height
        guard let image = ctx.makeImage() else { return nil }
        return CIImage(cgImage: image)
    }
}

class ColumnMask {
    enum Edge { case top, bottom }
    var currentEdge: Edge = .bottom
    var padding: CGFloat = 0
    var bitmapBytes = 0
    func verticalGradient(size: CGSize) -> CIImage? {
        let width = Int(round(size.width))
        guard width > 0 else { return nil }
        guard let ctx = CGContext(
            data: nil,
            width: 1,
            height: Int(round(size.height)),
            bitsPerComponent: 8,
            bytesPerRow: 0,
            space: CGColorSpaceCreateDeviceGray(),
            bitmapInfo: CGImageAlphaInfo.none.rawValue
        ) else { return nil }

        let colors: [NSColor]
        switch currentEdge {
        case .bottom:
            colors = [.white, .black]
        case .top:
            colors = [.black, .white]
        }
        let cgcolors = colors.map { $0.cgColor } as CFArray

        guard let gradient = CGGradient(
            colorsSpace: CGColorSpaceCreateDeviceGray(),
            colors: cgcolors,
            locations: [0.0, 1.0]
        ) else { return nil }

        ctx.drawLinearGradient(
            gradient,
            start: CGPoint(x: 0, y: -padding),
            end: CGPoint(x: 0, y: size.height + padding),
            options: []
        )

        bitmapBytes = ctx.bytesPerRow * ctx.height
        guard let image = ctx.makeImage() else { return nil }
        return CIImage(cgImage: image).clampedToExtent()
            .transformed(by: CGAffineTransform(scaleX: CGFloat(width), y: 1))
            .cropped(to: CGRect(x: 0, y: 0, width: width, height: image.height))
    }
}
let context = CIContext(options: [.cacheIntermediates: false])
let colorSpace = CGColorSpace(name: CGColorSpace.sRGB)!
func pixels(_ image: CIImage, rect: CGRect) -> [UInt8] {
    var data = [UInt8](repeating: 0, count: Int(rect.width * rect.height) * 4)
    data.withUnsafeMutableBytes { context.render(image, toBitmap: $0.baseAddress!, rowBytes: Int(rect.width) * 4, bounds: rect, format: .RGBA8, colorSpace: colorSpace) }
    return data
}
let a = OriginalMask(), b = ColumnMask()
var maskCases = 0, filterCases = 0, differences = 0, sourceBytes: [[String: Int]] = []
var diagnostics: [[String:Any]] = []
for width: CGFloat in [1, 2, 17, 320, 640.25] {
    for height: CGFloat in [1, 2, 17.5, 40.25] {
        for padding: CGFloat in [-5, 0, 4.25] {
            for top in [false, true] {
                a.currentEdge = top ? .top : .bottom; b.currentEdge = top ? .top : .bottom
                a.padding = padding; b.padding = padding
                let size = CGSize(width: width, height: height)
                let x = a.verticalGradient(size: size)!, y = b.verticalGradient(size: size)!
                precondition(x.extent == y.extent)
                let xp = pixels(x, rect: x.extent), yp = pixels(y, rect: y.extent)
                if xp != yp { differences += 1; diagnostics.append(["width":width,"height":height,"padding":padding,"top":top,"max_difference":zip(xp,yp).map{abs(Int($0)-Int($1))}.max()!,"baseline_prefix":Array(xp.prefix(24)),"column_prefix":Array(yp.prefix(24))]) }
                maskCases += 1
                if height == 40.25 && padding == 0 && !top { sourceBytes.append(["width":Int(width),"baseline":a.bitmapBytes,"column":b.bitmapBytes]) }
                if width == 320 && height == 40.25 {
                    let input = CIFilter(name: "CICheckerboardGenerator")!
                    input.setValue(7, forKey: "inputWidth")
                    input.setValue(CIColor(red: 1, green: 0.1, blue: 0.2), forKey: "inputColor0")
                    input.setValue(CIColor(red: 0.1, green: 0.5, blue: 1), forKey: "inputColor1")
                    for radius in [0.0,8.0,20.0] {
                        let first = CIFilter(name: "CIMaskedVariableBlur")!, second = CIFilter(name: "CIMaskedVariableBlur")!
                        for f in [first,second] { f.setValue(input.outputImage!, forKey: kCIInputImageKey); f.setValue(radius, forKey:kCIInputRadiusKey) }
                        first.setValue(x,forKey:"inputMask");second.setValue(y,forKey:"inputMask")
                        let xp = pixels(first.outputImage!,rect:x.extent),yp = pixels(second.outputImage!,rect:y.extent)
                        precondition(xp.contains{$0 != 0})
                        if xp != yp { differences += 1; print("FILTER_DIFF",padding,top,radius,zip(xp,yp).filter{$0 != $1}.count) }
                        filterCases += 1
                    }
                }
            }
        }
    }
}
let result: [String:Any] = ["mask_cases":maskCases,"filtered_cases":filterCases,"different_cases":differences,"source_bitmap_bytes":sourceBytes,"diagnostics":diagnostics]
try JSONSerialization.data(withJSONObject:result, options:[.prettyPrinted,.sortedKeys]).write(to:URL(fileURLWithPath:"/tmp/blur-edge-mask-probe.json"))
print(result)
