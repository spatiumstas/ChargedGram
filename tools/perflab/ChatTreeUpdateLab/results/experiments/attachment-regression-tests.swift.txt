import Cocoa
import XCTest
import TGUIKit
@testable import ChatLayerEngine
@testable import TreeBaseline

private struct CountedID: Hashable {
    static var hashCalls = 0
    let value: Int
    func hash(into hasher: inout Hasher) { Self.hashCalls += 1; hasher.combine(value) }
}

final class TreeUpdateTests: XCTestCase {
    struct Pair {
        let old: TreeBaseline.ChatLayerSceneView
        let new: ChatLayerEngine.ChatLayerSceneView
        let oldLeft: TreeBaseline.ChatLayerNode
        let newLeft: ChatLayerEngine.ChatLayerNode
        let oldRight: TreeBaseline.ChatLayerNode
        let newRight: ChatLayerEngine.ChatLayerNode
        let a: [TreeBaseline.ChatLayerNode]
        let b: [ChatLayerEngine.ChatLayerNode]
    }

    override func setUp() {
        TGUIKitTestBootstrap.install()
        TGUIKitAccessibility.isEnabled = true
    }

    func pair(_ count: Int = 4, background: Int = 12) throws -> Pair {
        let frame = CGRect(x: 0, y: 0, width: 400, height: 200)
        let old = TreeBaseline.ChatLayerSceneView(frame: frame), new = ChatLayerEngine.ChatLayerSceneView(frame: frame)
        let oldLeft = TreeBaseline.ChatLayerNode(id: "left"), newLeft = ChatLayerEngine.ChatLayerNode(id: "left")
        let oldRight = TreeBaseline.ChatLayerNode(id: "right"), newRight = ChatLayerEngine.ChatLayerNode(id: "right")
        oldLeft.frame = frame; newLeft.frame = frame
        oldRight.frame = CGRect(x: 200, y: 0, width: 100, height: 100); newRight.frame = oldRight.frame
        oldLeft.layer.addSublayer(SimpleLayer()); newLeft.layer.addSublayer(SimpleLayer())
        let a = (0..<count).map { TreeBaseline.ChatLayerNode(id: "child\($0)") }
        let b = (0..<count).map { ChatLayerEngine.ChatLayerNode(id: "child\($0)") }
        for (index, node) in a.enumerated() {
            node.frame = CGRect(x: 10, y: 10, width: 60, height: 50)
            node.acceptsPointerEvents = true; node.acceptsKeyboardFocus = true
            node.accessibilityRole = .button; node.accessibilityLabel = "child\(index)"
            try node.setChildren([TreeBaseline.ChatLayerNode(id: "leaf\(index)")])
        }
        for (index, node) in b.enumerated() {
            node.frame = CGRect(x: 10, y: 10, width: 60, height: 50)
            node.acceptsPointerEvents = true; node.acceptsKeyboardFocus = true
            node.accessibilityRole = .button; node.accessibilityLabel = "child\(index)"
            try node.setChildren([ChatLayerEngine.ChatLayerNode(id: "leaf\(index)")])
        }
        try oldLeft.setChildren(a); try newLeft.setChildren(b)
        try oldRight.setChildren((0..<background).map { TreeBaseline.ChatLayerNode(id: "background\($0)") })
        try newRight.setChildren((0..<background).map { ChatLayerEngine.ChatLayerNode(id: "background\($0)") })
        try old.root.setChildren([oldLeft, oldRight]); try new.root.setChildren([newLeft, newRight])
        old.layout(); new.layout()
        return Pair(old: old, new: new, oldLeft: oldLeft, newLeft: newLeft, oldRight: oldRight, newRight: newRight, a: a, b: b)
    }

    func testEveryUniqueSubsetAndPermutationPreservesTreeAndLookup() throws {
        var orders: [[Int]] = []
        func visit(_ values: [Int]) {
            orders.append(values)
            for next in 0..<4 where !values.contains(next) { visit(values + [next]) }
        }
        visit([])
        XCTAssertEqual(orders.count, 65)
        for order in orders {
            let p = try pair()
            let oldContent = p.oldLeft.layer.sublayers!.first!, newContent = p.newLeft.layer.sublayers!.first!
            try p.oldLeft.setChildren(order.map { p.a[$0] }); try p.newLeft.setChildren(order.map { p.b[$0] })
            XCTAssertEqual(p.oldLeft.children.map(\.id), p.newLeft.children.map(\.id))
            XCTAssertTrue(p.oldLeft.layer.sublayers?.first === oldContent)
            XCTAssertTrue(p.newLeft.layer.sublayers?.first === newContent)
            for (position, index) in order.enumerated() {
                XCTAssertTrue(p.oldLeft.layer.sublayers?[position + 1] === p.a[index].layer)
                XCTAssertTrue(p.newLeft.layer.sublayers?[position + 1] === p.b[index].layer)
            }
            for index in 0..<4 {
                for id in ["child\(index)", "leaf\(index)"] {
                    XCTAssertEqual(p.old.node(id: id)?.id, p.new.node(id: id)?.id)
                    XCTAssertEqual(p.new.node(id: id) != nil, order.contains(index))
                }
                XCTAssertEqual(p.a[index].parent?.id, p.b[index].parent?.id)
                XCTAssertEqual(p.b[index].scene === p.new, order.contains(index))
            }
            for index in 0..<12 { XCTAssertNotNil(p.new.node(id: "background\(index)")) }
            XCTAssertEqual(p.old.hitNode(at: CGPoint(x: 20, y: 20))?.id, p.new.hitNode(at: CGPoint(x: 20, y: 20))?.id)
            XCTAssertEqual(p.old.accessibilityChildren()?.count, p.new.accessibilityChildren()?.count)
        }
    }

    func testInvalidChangesKeepOriginalErrorsAndLeaveTreeUntouched() throws {
        for kind in 0..<6 {
            let p = try pair()
            let a: [TreeBaseline.ChatLayerNode], b: [ChatLayerEngine.ChatLayerNode]
            switch kind {
            case 0: a = [p.a[0], p.a[0]]; b = [p.b[0], p.b[0]]
            case 1: a = [TreeBaseline.ChatLayerNode(id: "background0")]; b = [ChatLayerEngine.ChatLayerNode(id: "background0")]
            case 2: a = [p.old.root]; b = [p.new.root]
            case 3: a = [p.oldRight.children[0]]; b = [p.newRight.children[0]]
            case 4: a = [TreeBaseline.ChatLayerNode(id: "fresh"), TreeBaseline.ChatLayerNode(id: "fresh")]; b = [ChatLayerEngine.ChatLayerNode(id: "fresh"), ChatLayerEngine.ChatLayerNode(id: "fresh")]
            default: a = [p.a[0], p.a[0], p.oldRight.children[0]]; b = [p.b[0], p.b[0], p.newRight.children[0]]
            }
            var oldError: String?, newError: String?
            do { try p.oldLeft.setChildren(a) } catch { oldError = String(describing: error) }
            do { try p.newLeft.setChildren(b) } catch { newError = String(describing: error) }
            XCTAssertNotNil(oldError); XCTAssertEqual(oldError, newError)
            XCTAssertEqual(p.newLeft.children.map(\.id), p.b.map(\.id))
            XCTAssertEqual(p.oldLeft.children.map(\.id), p.a.map(\.id))
            for index in 0..<4 { XCTAssertTrue(p.new.node(id: "child\(index)") === p.b[index]) }
        }
    }

    func testNewReplacementAndDetachedReparentingRefreshLookup() throws {
        let p = try pair()
        let a = TreeBaseline.ChatLayerNode(id: "child0"), b = ChatLayerEngine.ChatLayerNode(id: "child0")
        try p.oldLeft.setChildren([a] + Array(p.a.dropFirst())); try p.newLeft.setChildren([b] + Array(p.b.dropFirst()))
        XCTAssertTrue(p.old.node(id: "child0") === a); XCTAssertTrue(p.new.node(id: "child0") === b)
        XCTAssertNil(p.old.node(id: "leaf0")); XCTAssertNil(p.new.node(id: "leaf0"))
        try p.oldLeft.setChildren([]); try p.newLeft.setChildren([])
        try p.oldRight.setChildren(p.oldRight.children + [a]); try p.newRight.setChildren(p.newRight.children + [b])
        XCTAssertTrue(p.new.node(id: "child0") === b)
        XCTAssertTrue(b.parent === p.newRight)
        XCTAssertTrue(b.scene === p.new)
        XCTAssertEqual(p.oldRight.children.map(\.id), p.newRight.children.map(\.id))
    }

    func testPureReorderDoesNotRehashNodeIdentifiers() throws {
        let frame = CGRect(x: 0, y: 0, width: 400, height: 200)
        let old = TreeBaseline.ChatLayerSceneView(frame: frame), new = ChatLayerEngine.ChatLayerSceneView(frame: frame)
        let a = (0..<64).map { TreeBaseline.ChatLayerNode(id: CountedID(value: $0)) }
        let b = (0..<64).map { ChatLayerEngine.ChatLayerNode(id: CountedID(value: $0)) }
        try old.root.setChildren(a); try new.root.setChildren(b)
        CountedID.hashCalls = 0
        try old.root.setChildren(Array(a.reversed()))
        let oldCalls = CountedID.hashCalls
        CountedID.hashCalls = 0
        try new.root.setChildren(Array(b.reversed()))
        XCTAssertGreaterThan(oldCalls, 128)
        XCTAssertEqual(CountedID.hashCalls, 0)
        for index in 0..<64 { XCTAssertTrue(new.node(id: CountedID(value: index)) === b[index]) }
        XCTAssertThrowsError(try new.root.setChildren(b + [ChatLayerEngine.ChatLayerNode(id: CountedID(value: 0))]))
    }

    func testHoverFocusAndCaptureMatchAcrossReorderAndRemoval() throws {
        let p = try pair()
        var a = [String](), b = [String]()
        for node in p.a { let id = node.id; node.onPointer = { a.append("\(id):\($0.kind)"); return $0.kind == .down } }
        for node in p.b { let id = node.id; node.onPointer = { b.append("\(id):\($0.kind)"); return $0.kind == .down } }
        let point = CGPoint(x: 20, y: 20)
        _ = p.old.handlePointer(.moved, at: point); _ = p.new.handlePointer(.moved, at: point)
        _ = p.old.handlePointer(.down, at: point); _ = p.new.handlePointer(.down, at: point)
        try p.oldLeft.setChildren(Array(p.a.reversed())); try p.newLeft.setChildren(Array(p.b.reversed()))
        _ = p.old.handlePointer(.up, at: point); _ = p.new.handlePointer(.up, at: point)
        XCTAssertEqual(a, b); XCTAssertTrue(b.contains("child3:up"))
        XCTAssertEqual(p.old.focusedNode?.id, p.new.focusedNode?.id)
        _ = p.old.handlePointer(.down, at: point); _ = p.new.handlePointer(.down, at: point)
        try p.oldLeft.setChildren(Array(p.a.dropFirst())); try p.newLeft.setChildren(Array(p.b.dropFirst()))
        XCTAssertEqual(a, b); XCTAssertTrue(b.contains("child0:cancelled"))
        XCTAssertNil(p.new.focusedNode)
    }

    func testReentrantHoverMutationStillPublishesNewNodeLookup() throws {
        let p = try pair()
        let a = TreeBaseline.ChatLayerNode(id: "reentrant"), b = ChatLayerEngine.ChatLayerNode(id: "reentrant")
        p.a[0].onPointer = { [weak parent = p.oldRight] event in
            if event.kind == .entered, let parent, a.parent == nil { try! parent.setChildren(parent.children + [a]) }
            return false
        }
        p.b[0].onPointer = { [weak parent = p.newRight] event in
            if event.kind == .entered, let parent, b.parent == nil { try! parent.setChildren(parent.children + [b]) }
            return false
        }
        let point = CGPoint(x: 20, y: 20)
        _ = p.old.handlePointer(.moved, at: point); _ = p.new.handlePointer(.moved, at: point)
        try p.oldLeft.setChildren(Array(p.a.reversed())); try p.newLeft.setChildren(Array(p.b.reversed()))
        XCTAssertTrue(p.old.node(id: "reentrant") === a); XCTAssertTrue(p.new.node(id: "reentrant") === b)
        XCTAssertEqual(p.oldRight.children.map(\.id), p.newRight.children.map(\.id))
        XCTAssertEqual(p.old.hitNode(at: point)?.id, p.new.hitNode(at: point)?.id)
    }

    func testReentrantAttachmentReorderPublishesLookupBeforeOuterUpdateReturns() throws {
        let p = try pair()
        let a = TreeBaseline.ChatLayerNode(id: "attaching"), b = ChatLayerEngine.ChatLayerNode(id: "attaching")
        var oldPublished = false, newPublished = false
        a.onWindowKeyChanged = { [weak parent = p.oldLeft, weak scene = p.old] _ in
            guard let parent, let scene else { return }
            try! parent.setChildren(Array(parent.children.reversed()))
            oldPublished = scene.node(id: "attaching") === a
        }
        b.onWindowKeyChanged = { [weak parent = p.newLeft, weak scene = p.new] _ in
            guard let parent, let scene else { return }
            try! parent.setChildren(Array(parent.children.reversed()))
            newPublished = scene.node(id: "attaching") === b
        }
        try p.oldLeft.setChildren(p.a + [a]); try p.newLeft.setChildren(p.b + [b])
        XCTAssertTrue(oldPublished)
        XCTAssertTrue(newPublished)
        XCTAssertEqual(p.oldLeft.children.map(\.id), p.newLeft.children.map(\.id))
        XCTAssertTrue(p.old.node(id: "attaching") === a)
        XCTAssertTrue(p.new.node(id: "attaching") === b)
        a.onWindowKeyChanged = nil; b.onWindowKeyChanged = nil
    }

    func testRemovedSubtreesAreReleased() throws {
        let scene = ChatLayerEngine.ChatLayerSceneView(frame: CGRect(x: 0, y: 0, width: 100, height: 100))
        weak var parent: ChatLayerEngine.ChatLayerNode?, child: ChatLayerEngine.ChatLayerNode?
        try autoreleasepool {
            let p = ChatLayerEngine.ChatLayerNode(id: "parent"), c = ChatLayerEngine.ChatLayerNode(id: "child")
            parent = p; child = c
            try p.setChildren([c]); try scene.root.setChildren([p])
            try scene.root.setChildren([])
            XCTAssertNil(scene.node(id: "parent")); XCTAssertNil(scene.node(id: "child"))
        }
        XCTAssertNil(parent); XCTAssertNil(child)
    }
}
